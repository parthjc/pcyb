def load_api_keys():
    try:
        if os.path.exists(API_KEYS_FILE):
            with open(API_KEYS_FILE, 'r') as f:
                data = json.load(f)
                if 'keys' not in data:
                    data['keys'] = {}
                for k in ['key1', 'key2', 'key3']:
                    if k not in data['keys']:
                        data['keys'][k] = ""
                if 'active' not in data:
                    data['active'] = 'key1'
                if 'key_names' not in data:
                    data['key_names'] = {
                        "key1": "Primary Key",
                        "key2": "Backup Key 1",
                        "key3": "Backup Key 2"
                    }
                return data
    except:
        pass

def save_api_keys(data):

        data['last_updated'] = datetime.now().isoformat()

    except:

def validate_indian_rc(rc: str) -> bool:
    state_codes = [
        "AP","AR","AS","BR","CG","GA","GJ","HR","HP","JK","JH",
        "KA","KL","MP","MH","MN","ML","MZ","NL","OR","PB","RJ",
        "SK","TN","TR","UP","WB","AN","CH","DN","DD","DL","LD",
        "PY","TS","TG","LA"
    ]
    pattern = r'^(' + '|'.join(state_codes) + r')[-]?\d{1,2}[-]?[A-Z]{1,2}[-]?\d{4}$'
    return re.match(pattern, rc) is not None

def fetch_from_api(rc: str, api_config: Dict) -> Dict:
    try:
        params = {}
        data = None
        json_data = None

        if api_config.get("params_mapping"):
            for param_key, param_value in api_config["params_mapping"].items():
                if param_value is None:
                    params[param_key] = rc
                else:
                    params[param_key] = param_value

        if api_config.get("is_form"):
            data = params
            params = {}

        if api_config["method"] == "POST" and not api_config.get("is_form"):
            json_data = params
            params = {}

        if api_config["method"] == "GET":
            response = requests.get(
                api_config["url"],
                params=params,
                headers=api_config.get("headers", {}),
                timeout=15
            )
        elif api_config["method"] == "POST":
            if api_config.get("is_form"):
                response = requests.post(
                    api_config["url"],
                    data=data,
                    headers=api_config.get("headers", {}),
                    timeout=15
                )
            else:
                response = requests.post(
                    api_config["url"],
                    json=json_data,
                    headers=api_config.get("headers", {}),
                    timeout=15
                )
        else:
            return {"error": f"Unsupported method: {api_config['method']}"}

        if response.status_code == 200:
            try:
                return response.json()
            except:
                return {"raw_text": response.text, "status": "non-json"}
        return {"error": f"HTTP {response.status_code}"}
    except Exception as e:
        return {"error": str(e)}

def try_with_all_keys(rc: str, api_config: Dict) -> Dict:
    valid_keys = get_all_valid_keys()
    if not valid_keys:
        return {"error": "No API keys configured!"}
    
    for key_id, key_value in valid_keys:
        api_copy = api_config.copy()
        if 'headers' in api_copy and 'x-rapidapi-key' in api_copy['headers']:
            api_copy['headers']['x-rapidapi-key'] = key_value
        
        result = fetch_from_api(rc, api_copy)
        
        if "error" not in result and len(result) > 0:
            result['_key_used'] = key_id
            return result
    
    return {"error": "All keys failed"}

def lookup_rc_with_options(rc: str, mode: str = "auto", priority_range: str = "1-25") -> Dict:
    rc = re.sub(r'[-]', '', rc).upper()
    
    valid_keys = get_all_valid_keys()
    if not valid_keys:
        return {"error": "No API keys configured!"}
    
    all_apis = get_apis_with_key(get_active_key())
    selected_apis = []
    
    if mode == "auto":
        selected_apis = all_apis
    elif mode == "top5":
        selected_apis = [api for api in all_apis if api.get('priority', 0) <= 5]
    elif mode == "top10":
        selected_apis = [api for api in all_apis if api.get('priority', 0) <= 10]
    elif mode == "top15":
        selected_apis = [api for api in all_apis if api.get('priority', 0) <= 15]
    elif mode == "range":
        try:
            parts = priority_range.split('-')
            start = int(parts[0])
            end = int(parts[1])
            selected_apis = [api for api in all_apis if start <= api.get('priority', 0) <= end]
        except:
            selected_apis = all_apis
    elif mode == "single":
        try:
            p = int(priority_range)
            selected_apis = [api for api in all_apis if api.get('priority', 0) == p]
        except:
            selected_apis = all_apis
    else:
        selected_apis = all_apis
    
    if not selected_apis:
        return {"error": "No APIs selected for this mode!"}
    
    for api_config in selected_apis:
        result = try_with_all_keys(rc, api_config)
        
        if "error" not in result and len(result) > 0:
            key_used = result.pop('_key_used', 'unknown')
            return {
                "data": result,
                "api": api_config.get('name', 'Unknown'),
                "priority": api_config.get('priority', 0),
                "key_used": key_used,
                "total_apis": len(selected_apis)
            }
    
    return {"error": f"All {len(selected_apis)} APIs failed!"}

def format_vehicle_info(data: Dict, rc: str, api_name: str, priority: int, key_used: str = "", total_apis: int = 0) -> str:
    """Format vehicle info - PLAIN TEXT (no Markdown)"""
    
    state = "Unknown"
    if rc:
        clean_rc = re.sub(r'[-]', '', rc)
        state = clean_rc[:2].upper() if len(clean_rc) >= 2 else "Unknown"
    
    result = f"🚗 VEHICLE INFORMATION - {rc}\n"
    result += f"📍 State: {state}\n"
    result += f"📡 API: {api_name} (Priority #{priority})\n"
    if total_apis:
        result += f"📊 APIs Tried: {total_apis}\n"
    if key_used:
        result += f"🔑 Key Used: {key_used}\n"
    result += "─" * 40 + "\n"
    
    if "error" in data:
        result += f"❌ Error: {data['error']}"
        return result
    
    field_labels = {
        'owner_name': '👤 Owner Name', 'owner': '👤 Owner Name', 'name': '👤 Owner Name',
        'father_name': '👨‍👦 Father Name', 'address': '📍 Address', 'city': '🏙️ City',
        'state': '📍 State', 'state_name': '📍 State', 'country': '🌍 Country',
        'phone': '📞 Phone Number', 'mobile': '📞 Phone Number', 'email': '📧 Email',
        'rto': '🏢 RTO Office', 'registration_no': '📋 Registration No',
        'vehicle_no': '📋 Registration No', 'rc_number': '📋 Registration No',
        'registration_date': '📅 Registration Date', 'vehicle_class': '🚗 Vehicle Class',
        'maker': '🏭 Maker', 'manufacturer': '🏭 Maker', 'model': '🔧 Model',
        'fuel_type': '⛽ Fuel Type', 'engine_no': '🔢 Engine Number',
        'chassis_no': '🔢 Chassis Number', 'insurance_company': '🛡️ Insurance Company',
        'insurance_upto': '📆 Insurance Upto', 'puc_valid_upto': '📋 PUC Valid Upto',
        'road_tax_upto': '💰 Road Tax Upto', 'registration_status': '📊 Registration Status',
        'color': '🎨 Color', 'manufactured_year': '🏗️ Manufactured Year'
    }
    
    displayed = 0
    for key, label in field_labels.items():
        if key in data and data[key]:
            # Clean value - remove special characters that break formatting
            value = str(data[key]).replace('*', '').replace('_', '').replace('`', '').replace('[', '').replace(']', '')
            result += f"{label}: {value}\n"
            displayed += 1
    
    if displayed == 0:
        result += "📄 Raw Data:\n"
        raw_json = json.dumps(data, indent=2, ensure_ascii=False)
        if len(raw_json) > 500:
            result += raw_json[:500] + "...\n(truncated)"
        else:
            result += raw_json
    
    result += "\n" + "─" * 40 + "\n"
    result += f"🔒 {COPYRIGHT}"
    
    return result

    query = update.callback_query
    await query.answer()

        await query.edit_message_text("❌ Unauthorized.")
        return

        await query.edit_message_text("❌ Cancelled. Send /search to try again.")
        return
    
    if data == "back_to_options":
        await query.edit_message_text(
            "🔍 Select Search Mode\n\nThen send your RC number.",
            reply_markup=get_search_options_keyboard()
        )
        return
    
    if data.startswith("mode_"):
        mode = data.replace("mode_", "")
        context.user_data['search_mode'] = mode
        context.user_data['search_range'] = "1-25"
        
        mode_names = {
            "auto": "🚀 AUTO (All 25 APIs)",
            "top5": "⭐ TOP 5 (Best 5 APIs)",
            "top10": "🌟 TOP 10 (Best 10 APIs)",
            "top15": "💫 TOP 15 (Best 15 APIs)",
            "single": "📌 SINGLE API (Select below)"
        }
        
        msg = f"✅ Mode selected: {mode_names.get(mode, mode)}\n\n"
        
        if mode == "single":
            msg += "👇 Select an API priority number:\n"
            await query.edit_message_text(
                msg,
                reply_markup=get_api_list_keyboard()
            )
        else:
            msg += "📝 Now send your RC number:"
            await query.edit_message_text(msg)
        
        return
    
    if data.startswith("range_"):
        range_val = data.replace("range_", "")
        context.user_data['search_mode'] = "range"
        context.user_data['search_range'] = range_val
        
        msg = f"✅ Range selected: {range_val}\n\n📝 Now send your RC number:"
        await query.edit_message_text(msg)
        return
    
    if data.startswith("api_"):
        priority = data.replace("api_", "")
        context.user_data['search_mode'] = "single"
        context.user_data['search_range'] = priority
        
        msg = f"✅ API selected: Priority #{priority}\n\n📝 Now send your RC number:"
        await query.edit_message_text(msg)
        return

async def testkeys_command(update: Update, context: ContextTypes.DEFAULT_TYPE):
    user_id = update.effective_user.id
    if ALLOWED_USERS and user_id not in ALLOWED_USERS:
        await update.message.reply_text("❌ Unauthorized.")
        return
    
    await update.message.reply_text("🔄 Testing all keys... This may take a moment.")
    
    keys_data, active = get_keys_with_status()
    results = []
    
    for key_id, info in keys_data.items():
        key_value = info['key']
        if not key_value:
            results.append(f"{key_id}: ⚠️ EMPTY")
            continue
        
        try:
            test_url = "https://vahan-vehicle-details-api.p.rapidapi.com/rc_v2.php"
            test_params = {"registration_no": "GJ05VC1234"}
            headers = {
                "x-rapidapi-host": "vahan-vehicle-details-api.p.rapidapi.com",
                "x-rapidapi-key": key_value
            }
            response = requests.get(test_url, params=test_params, headers=headers, timeout=10)
            
            if response.status_code in [200, 404, 429]:
                results.append(f"{key_id}: ✅ VALID ({response.status_code})")
            else:
                results.append(f"{key_id}: ❌ INVALID ({response.status_code})")
        except Exception as e:
            results.append(f"{key_id}: ❌ ERROR - {str(e)[:30]}")
    
    msg = "🧪 Key Test Results\n\n" + "\n".join(results)
    await update.message.reply_text(msg)
