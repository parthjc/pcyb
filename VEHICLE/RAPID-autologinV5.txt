def install_package(package):
    try:
        subprocess.check_call([sys.executable, '-m', 'pip', 'install', package])
        print(f"✅ Installed: {package}")
        return True
    except Exception as e:
        print(f"❌ Failed to install {package}: {e}")
        return False

def ensure_requirements():
    required = ['playwright', 'requests', 'setuptools']
    for pkg in required:
        try:
            importlib.import_module(pkg)
        except ImportError:
            print(f"⚠️ {pkg} not found. Installing...")
            install_package(pkg)
    try:
        subprocess.check_call([sys.executable, '-m', 'playwright', 'install', 'chromium'])
        print(f"✅ Playwright Chromium Installed.")
    except:
        pass

# ===================================================================
# 💾 SAVE & LOAD CREDENTIALS
# ===================================================================
def load_credentials():
    if os.path.exists(CREDENTIALS_FILE):
        try:
            with open(CREDENTIALS_FILE, 'r') as f:
                return json.load(f)
        except:
            return []
    return []

def save_credentials(creds_list):
    with open(CREDENTIALS_FILE, 'w') as f:
        json.dump(creds_list, f, indent=4)

# 🧹 DYNAMIC POPUP HANDLER
# ===================================================================
async def handle_popups(page):
    try:
        # 1. "Save password?" popup -> SAVE KARO
        save_btn = page.locator("button:has-text('Save')").first
        if await save_btn.is_visible():
            await save_btn.click()
            await page.wait_for_timeout(1000)
            print("   🔑 Save password popup: SAVED!")
            return True
        # 2. "No thanks" / "Never" button -> HATAVO
        no_thanks_btn = page.locator("button:has-text('No thanks'), button:has-text('Never')").first
        if await no_thanks_btn.is_visible():
            await no_thanks_btn.click()
            await page.wait_for_timeout(1000)
            print("   ❌ Popup removed (No thanks)!")
        # 3. Cookie popup -> ACCEPT KARO
        cookie_btn = page.locator("button:has-text('Accept All'), button:has-text('Reject All')").first
        if await cookie_btn.is_visible():
            await cookie_btn.click()
            await page.wait_for_timeout(1000)
            print("   🍪 Cookie popup cleared!")
        # 4. Generic Close button -> HATAVO
        close_btn = page.locator("button[aria-label='Close'], button:has-text('Close')").first
        if await close_btn.is_visible():
            await close_btn.click()
            await page.wait_for_timeout(1000)
            print("   ❌ Generic popup cleared!")
    except:
        pass
    return False

# ===================================================================
# ⚠️ "Something Went Wrong" ERROR PAGE HANDLE KARO
# ===================================================================
async def handle_error_page(page, search_url):
    try:
        error_text = page.locator("text=Something Went Wrong").first
        if await error_text.is_visible():
            print("   ⚠️ Something Went Wrong page detected!")
            try:
                try_again_btn = page.locator("button:has-text('Try again')").first
                if await try_again_btn.is_visible():
                    await try_again_btn.click()
                    await page.wait_for_timeout(5000)
                    print("   ✅ Try again button clicked!")
                    return True
            except:
                pass
            print("   ⚠️ Try again failed. Going back to search page...")
            await page.goto(search_url, wait_until="commit")
            await page.wait_for_timeout(8000)
            return True
    except:
        pass
    return False

# 🚀 MAIN SUBSCRIBE FUNCTION
# ===================================================================
async def subscribe_all_apis():
    async with async_playwright() as p:
        user_data_dir = "./user_data"
        context = await p.chromium.launch_persistent_context(
            user_data_dir, 
            headless=False,
            viewport={'width': 1280, 'height': 720}
        )
        page = context.pages[0] if context.pages else await context.new_page()
        page.set_default_timeout(60000)

        print(f"{Colors.CYAN}  🔒 Starting Subscription Process...{Colors.END}")

        print("   🔍 Checking page status...")
        try:
            await page.goto("https://rapidapi.com/search?term=rto", wait_until="commit")
        except:
            pass
        await page.wait_for_timeout(15000)

        await handle_popups(page)

        print("   🔍 Checking if login required...")
        try:
            sign_in_btn = page.locator("button:has-text('Sign In'), a:has-text('Sign In')").first
            if await sign_in_btn.is_visible():
                print("   ⚠️ Sign In required! Opening login page...")
                await page.goto("https://rapidapi.com/auth/login", wait_until="commit")
                await page.wait_for_timeout(7000)
                await handle_popups(page)
                try:
                    await page.locator("#email").wait_for(state="visible", timeout=30000)
                    await page.locator("#email").fill(EMAIL)
                    await page.locator("#password").fill(PASSWORD)
                    await page.locator("button[type='submit']").click()
                    await page.wait_for_timeout(10000)
                    print("   ✅ Login successful!")
                except Exception as e:
                    print(f"   ⚠️ Login issue: {e}")
                    try:
                        await page.locator("input[type='email']").wait_for(state="visible", timeout=30000)
                        await page.locator("input[type='email']").fill(EMAIL)
                        await page.locator("input[type='password']").fill(PASSWORD)
                        await page.locator("button[type='submit']").click()
                        await page.wait_for_timeout(10000)
                        print("   ✅ Login successful (Fallback)!")
                    except Exception as e2:
                        print(f"   ❌ Login failed completely: {e2}")
                        return
                await page.goto("https://rapidapi.com/search?term=rto", wait_until="commit")
                await page.wait_for_timeout(15000)
                await handle_popups(page)
            else:
                print("   ✅ Already logged in! Skipping login...")
        except:
            print("   ✅ No Sign In button found. Already logged in!")

        print(f"\n🔍 Searching for '{SEARCH_TERM}'...")
        search_url = f"https://rapidapi.com/search?term={SEARCH_TERM}"
        try:
            await page.goto(search_url, wait_until="commit")
        except:
            pass
        await page.wait_for_timeout(15000)

        print("   ⬇️ Moving mouse cursor down (to center)...")
        await page.mouse.move(900, 500)
        await page.wait_for_timeout(2000)

        print("   ⬇️ Scrolling down (15 times)...")
        for _ in range(15):
            await page.mouse.wheel(0, 3000)
            await page.wait_for_timeout(1000)

        print("   🔢 Counting ALL APIs...")
        all_api_links = page.locator("a[href*='/api/']")
        total_count = await all_api_links.count()
        print(f"   ✅ Total {total_count} APIs found on the page.\n")

        total_subscribed = 0
        failed_apis = []

        for idx in range(total_count):
            print(f"   🖱️ Clicking API {idx+1} of {total_count}...")
            try:
                all_api_links = page.locator("a[href*='/api/']")
                await all_api_links.nth(idx).click()
                await page.wait_for_timeout(8000)
                await handle_error_page(page, search_url)
                await handle_popups(page)
                try:
                    subscribe_run_btn = page.locator("button:has-text('Subscribe to Run')").first
                    await subscribe_run_btn.wait_for(state="visible", timeout=10000)
                    await subscribe_run_btn.click()
                    print(f"      ✅ Clicked 'Subscribe to Run'")
                except:
                    change_plan_btn = page.locator("button:has-text('Change plan')").first
                    await change_plan_btn.wait_for(state="visible", timeout=10000)
                    await change_plan_btn.click()
                    print(f"      ✅ Clicked 'Change plan' (Already Subscribed)")
                await page.wait_for_timeout(5000)
                await handle_error_page(page, search_url)
                await handle_popups(page)
                try:
                    basic_button = page.locator("button:has-text('Start Free Plan'), button:has-text('Request Access')").first
                    await basic_button.wait_for(state="visible", timeout=20000)
                    await basic_button.click()
                    print(f"      ✅ Started Free Basic Plan!")
                except:
                    print(f"      ⚠️ No Free Plan available. Skipping...")
                    failed_apis.append(f"API {idx+1} (No Free Plan)")
                    await page.goto(search_url, wait_until="commit")
                    await page.wait_for_timeout(8000)
                    continue
                await page.wait_for_timeout(3000)
                confirm_button = page.locator("button:has-text('OK'), button:has-text('Confirm'), button:has-text('Subscribe')").first
                if await confirm_button.is_visible():
                    await confirm_button.click()
                    print(f"      🎉 SUBSCRIBED! (Total: {total_subscribed + 1})")
                    total_subscribed += 1
                await page.goto(search_url, wait_until="commit")
                await page.wait_for_timeout(8000)
                await handle_popups(page)
                print(f"   ⬇️ Moving mouse cursor down (to load next API)...")
                await page.mouse.move(900, 500)
                await page.wait_for_timeout(2000)
                print(f"   ⬇️ Scrolling down 2-3 times...")
                for _ in range(3):
                    await page.mouse.wheel(0, 3000)
                    await page.wait_for_timeout(1000)
            except Exception as e:
                print(f"      ❌ Error: {e}")
                failed_apis.append(f"API {idx+1} ({e})")
                await page.goto(search_url, wait_until="commit")
                await page.wait_for_timeout(8000)
                continue

        print(f"\n📊 FINAL REPORT:")
        print(f"   ✅ Successfully Subscribed: {total_subscribed} APIs")
        print(f"   ❌ Failed APIs: {len(failed_apis)}")
        if failed_apis:
            for api in failed_apis:
                print(f"      - {api}")
        print(f"\n🖥️ Browser 30 second open rahese - Check karo ke badhi subscribed chhe ke nahi.")
        await page.wait_for_timeout(30000)
        await context.close()
        print("Browser closed.")
