# 📊 SUBSCRIPTION LOG MANAGEMENT
# ===================================================================
def load_subscription_log():
    if os.path.exists(SUBSCRIPTION_LOG_FILE):
        try:
            with open(SUBSCRIPTION_LOG_FILE, 'r') as f:
                return json.load(f)
        except:
            return {"subscriptions": [], "last_run": None, "subscribed_apis": []}
    return {"subscriptions": [], "last_run": None, "subscribed_apis": []}

def save_subscription_log(log_data):
    with open(SUBSCRIPTION_LOG_FILE, 'w') as f:
        json.dump(log_data, f, indent=4)

def log_subscription(api_name, api_url, status="subscribed"):
    log = load_subscription_log()
    timestamp = datetime.now().isoformat()
    
    for sub in log["subscriptions"]:
        if sub["api_url"] == api_url:
            sub["status"] = status
            sub["last_checked"] = timestamp
            sub["email"] = EMAIL
            save_subscription_log(log)
            return
    
    log["subscriptions"].append({
        "api_name": api_name,
        "api_url": api_url,
        "status": status,
        "subscribed_at": timestamp,
        "last_checked": timestamp,
        "email": EMAIL
    })
    log["last_run"] = timestamp
    save_subscription_log(log)

def log_api_attempt(api_name, api_url, status="attempted"):
    log = load_subscription_log()
    timestamp = datetime.now().isoformat()
    
    if "attempted_apis" not in log:
        log["attempted_apis"] = []
    
    for api in log["attempted_apis"]:
        if api["api_url"] == api_url:
            api["status"] = status
            api["last_attempt"] = timestamp
            save_subscription_log(log)
            return
    
    log["attempted_apis"].append({
        "api_name": api_name,
        "api_url": api_url,
        "status": status,
        "attempted_at": timestamp,
        "email": EMAIL
    })
    save_subscription_log(log)

async def handle_popups(page):
    try:
        save_btn = page.locator("button:has-text('Save')").first
        if await save_btn.is_visible():
            await save_btn.click()
            await page.wait_for_timeout(1000)
            print("   🔑 Save password popup: SAVED!")
            return True
        no_thanks_btn = page.locator("button:has-text('No thanks'), button:has-text('Never')").first
        if await no_thanks_btn.is_visible():
            await no_thanks_btn.click()
            await page.wait_for_timeout(1000)
            print("   ❌ Popup removed (No thanks)!")
        cookie_btn = page.locator("button:has-text('Accept All'), button:has-text('Reject All')").first
        if await cookie_btn.is_visible():
            await cookie_btn.click()
            await page.wait_for_timeout(1000)
            print("   🍪 Cookie popup cleared!")
        close_btn = page.locator("button[aria-label='Close'], button:has-text('Close')").first
        if await close_btn.is_visible():
            await close_btn.click()
            await page.wait_for_timeout(1000)
            print("   ❌ Generic popup cleared!")
    except:
        pass
    return False

# ===================================================================
# 🔄 RELOAD PAGE IF ERROR OCCURS
# ===================================================================
async def handle_error_and_reload(page, search_url, max_retries=3):
    for attempt in range(max_retries):
        try:
            error_locator = page.locator("text=Something Went Wrong").first
            if await error_locator.is_visible():
                print(f"   ⚠️ Something Went Wrong page detected! (Attempt {attempt + 1}/{max_retries})")
                
                try:
                    try_again_btn = page.locator("button:has-text('Try again')").first
                    if await try_again_btn.is_visible():
                        await try_again_btn.click()
                        await page.wait_for_timeout(5000)
                        if not await page.locator("text=Something Went Wrong").first.is_visible():
                            print("   ✅ Error resolved after clicking Try again!")
                            return True
                except:
                    pass
                
                print(f"   🔄 Reloading page... (Attempt {attempt + 1}/{max_retries})")
                await page.reload(wait_until="domcontentloaded")
                await page.wait_for_timeout(8000)
                
                if not await page.locator("text=Something Went Wrong").first.is_visible():
                    print(f"   ✅ Error resolved after reload! (Attempt {attempt + 1})")
                    return True
                else:
                    if attempt == max_retries - 1:
                        print("   🔄 Last attempt: Going back to search page...")
                        await page.goto(search_url, wait_until="domcontentloaded")
                        await page.wait_for_timeout(8000)
                        await page.reload(wait_until="domcontentloaded")
                        await page.wait_for_timeout(5000)
                        if not await page.locator("text=Something Went Wrong").first.is_visible():
                            print("   ✅ Error resolved by going to search page!")
                            return True
        except Exception as e:
            print(f"   ⚠️ Error handling attempt {attempt + 1} failed: {e}")
            await page.wait_for_timeout(2000)
    
    print(f"   ❌ Could not resolve error after {max_retries} attempts.")
    return False

# 🔄 DYNAMIC SCROLLING FUNCTION - ENHANCED
# ===================================================================
async def dynamic_scroll_to_load_all(page, max_scrolls=50, wait_time=800):
    """Dynamically scroll to load all APIs with detection"""
    print("   ⬇️ Dynamic scrolling to load all APIs...")
    
    previous_count = 0
    same_count_streak = 0
    total_scrolled = 0
    
    for scroll_num in range(max_scrolls):
        # Scroll down
        await page.mouse.wheel(0, 2000)
        await page.wait_for_timeout(wait_time)
        
        # Count current API links
        current_links = page.locator("a[href*='/api/']")
        current_count = await current_links.count()
        
        # Show progress
        if current_count > previous_count:
            print(f"   📊 Loaded {current_count} APIs so far...")
            previous_count = current_count
            same_count_streak = 0
        else:
            same_count_streak += 1
        
        total_scrolled += 1
        
        # If no new APIs loaded for 5 consecutive scrolls, stop
        if same_count_streak >= 5:
            print(f"   ✅ No new APIs loading. Stopped after {total_scrolled} scrolls.")
            break
        
        # Also check if we've reached the bottom
        try:
            # Check if "No more results" or similar text appears
            no_more = page.locator("text=No more results, text=End of results, text=No more APIs")
            if await no_more.first.is_visible():
                print(f"   ✅ Reached end of results. Stopped after {total_scrolled} scrolls.")
                break
        except:
            pass
    
    # Final count
    final_links = page.locator("a[href*='/api/']")
    final_count = await final_links.count()
    print(f"   ✅ Total {final_count} APIs loaded after {total_scrolled} scrolls.\n")
    
    return final_count

# 🔍 ENSURE API IS LOADED - SCROLL TO TARGET
# ===================================================================
async def ensure_api_loaded(page, target_index, search_url, max_attempts=5):
    """Scroll until the target API is loaded"""
    for attempt in range(max_attempts):
        # Get current count
        all_links = page.locator("a[href*='/api/']")
        current_count = await all_links.count()
        
        if target_index < current_count:
            print(f"   ✅ API #{target_index + 1} is loaded (Total: {current_count})")
            return True
        
        print(f"   🔄 Attempt {attempt + 1}: Need API #{target_index + 1}, currently have {current_count}...")
        
        # Scroll more
        for _ in range(5):
            await page.mouse.wheel(0, 2000)
            await page.wait_for_timeout(500)
        
        # Refresh links
        all_links = page.locator("a[href*='/api/']")
        current_count = await all_links.count()
        
        if target_index < current_count:
            print(f"   ✅ Loaded! API #{target_index + 1} now available (Total: {current_count})")
            return True
        
        # If still not loaded, try reloading the page
        print(f"   🔄 Reloading search page...")
        await page.goto(search_url, wait_until="domcontentloaded")
        await page.wait_for_timeout(5000)
        
        # Scroll again
        for _ in range(10):
            await page.mouse.wheel(0, 2000)
            await page.wait_for_timeout(500)
    
    print(f"   ❌ Could not load API #{target_index + 1} after {max_attempts} attempts")
    return False
