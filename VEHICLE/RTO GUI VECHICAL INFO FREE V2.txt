def flatten(x,p=""):
 out={}
 if isinstance(x,dict):
  for k,v in x.items():
   n=f"{p}_{k}" if p else str(k)
   if isinstance(v,(dict,list)):out.update(flatten(v,n))
   else:out[n]=v
 elif isinstance(x,list):
  for i,v in enumerate(x):
   n=f"{p}_{i}" if p else str(i)
   if isinstance(v,(dict,list)):out.update(flatten(v,n))
   else:out[n]=v
 return out

def is_sensitive(k):
 k=str(k).lower().replace("-","_")
 return any(w in k for w in SENSITIVE)

def safe_result(raw):
 out={}
 for k,v in flatten(raw).items():
  out[k.lower()]="[MASKED — sensitive field]" if is_sensitive(k) else v
 return out

def vehicle_image(query, manufacturer=None, model=None):
    """Return a real vehicle image for brand+model using several public image sources.
    The function deliberately does NOT trust the RTO legal manufacturer string as a model.
    """
    query = str(query or "").strip()
    manufacturer = str(manufacturer or "").strip()
    model = str(model or "").strip()

    def clean(v):
        v = re.sub(r"\([^)]*\)", " ", v or "", flags=re.I)
        v = re.sub(r"\b(MOTORCYCLE AND SCOOTER INDIA|MOTORCYCLE|SCOOTER|AUTOMOBILE|INDIA|PRIVATE|PUBLIC|LIMITED|LTD|PVT|CORP|CORPORATION|COMPANY)\b", " ", v, flags=re.I)
        return re.sub(r"\s+", " ", v).strip()

    # Normalize long RTO manufacturer names to the actual brand.
    brands = [
        "Royal Enfield","Maruti Suzuki","Mercedes-Benz","Mercedes Benz",
        "Land Rover","Ashok Leyland","Force Motors","Mahindra","Volkswagen",
        "Hero","Honda","Bajaj","TVS","Yamaha","Suzuki","Tata","Hyundai",
        "Kia","Toyota","Ola","Ather","KTM","BMW","Audi","Skoda","Nissan",
        "Renault","MG","Jeep","Isuzu","Volvo","Jawa","Aprilia","Piaggio",
        "Lexus","Porsche","Ducati","Triumph","Benelli","Harley Davidson"
    ]
    low = manufacturer.lower()
    brand = ""
    for b in sorted(brands, key=len, reverse=True):
        if b.lower() in low:
            brand = b
            break
    if not brand:
        brand = clean(manufacturer)

    clean_model = clean(model)
    if not clean_model or clean_model.lower() in {"n/a", "na", "unknown", "-"}:
        # If model was not separately supplied, remove the detected brand from q.
        clean_model = clean(query)
        if brand:
            clean_model = re.sub(r"\b" + re.escape(brand) + r"\b", " ", clean_model, flags=re.I)
            clean_model = re.sub(r"\s+", " ", clean_model).strip()

    # Vehicle-type words improve image search but are not used as identity tokens.
    body_type = "motorcycle"
    mlow = clean_model.lower()
    if any(x in mlow for x in ("scooter", "activa", "maestro", "jupiter", "access", "chetak", "dio")):
        body_type = "scooter"
    elif any(x in mlow for x in ("car", "sedan", "hatchback", "suv", "swift", "baleno", "creta", "nexon", "thar", "scorpio")):
        body_type = "car"

    search_query = " ".join(x for x in [brand, clean_model, body_type] if x).strip()
    if not search_query:
        search_query = query or "vehicle"

    model_tokens = [x for x in re.sub(r"[^a-z0-9]+", " ", clean_model.lower()).split() if len(x) >= 2]
    brand_tokens = [x for x in re.sub(r"[^a-z0-9]+", " ", brand.lower()).split() if len(x) >= 2]

    def relevance(title, text=""):
        hay = re.sub(r"[^a-z0-9]+", " ", (str(title) + " " + str(text)).lower())
        words = set(hay.split())
        score = 0
        for t in model_tokens:
            if t in words: score += 10
        for t in brand_tokens:
            if t in words: score += 5
        if model_tokens and " ".join(model_tokens) in hay: score += 20
        if brand and brand.lower() in hay: score += 8
        bad = {"person","portrait","painting","book","wine","flower","animal","aircraft","ship","train","map","logo","flag","pdf","cecil","dreeme"}
        if words & bad: score -= 50
        return score

    def valid_url(u):
        return bool(u and re.match(r"^https?://", u, re.I))

    candidates=[]

    # 1) Wikimedia Commons — stable public API.
    try:
        for sq in [search_query, f"{brand} {clean_model}".strip(), clean_model]:
            if not sq: continue
            api_url=("https://commons.wikimedia.org/w/api.php?action=query&format=json&origin=*"
                     "&generator=search&gsrnamespace=6&gsrlimit=40&gsrsearch="+quote(sq)+
                     "&prop=imageinfo&iiprop=url&iiurlwidth=1000")
            req=Request(api_url,headers={"User-Agent":"ZYNTRASEC-Vehicle-GUI/4.0"})
            with urlopen(req,timeout=12) as r:
                data=json.loads(r.read().decode("utf-8","replace"))
            for page in data.get("query",{}).get("pages",{}).values():
                info=(page.get("imageinfo") or [{}])[0]
                u=info.get("thumburl") or info.get("url")
                title=page.get("title","").replace("File:","").strip()
                if valid_url(u):
                    candidates.append((relevance(title),u,title,"Wikimedia Commons"))
    except Exception as e:
        print("[image] Wikimedia:",e)

    # 2) Bing Images HTML — no API key required. Use thumbnails/result URLs.
    try:
        import urllib.parse
        bing="https://www.bing.com/images/search?view=detailv2&iss=sbi&FORM=SBIHMP&sbisrc=ImgPicker&q="+quote(search_query)
        # Normal image search page is more reliable than detail mode.
        bing="https://www.bing.com/images/search?q="+quote(search_query)
        req=Request(bing,headers={"User-Agent":"Mozilla/5.0 (X11; Linux x86_64) AppleWebKit/537.36 Chrome/131 Safari/537.36"})
        with urlopen(req,timeout=12) as r:
            page=r.read().decode("utf-8","ignore")
        for m in re.finditer(r'<a[^>]+class="iusc"[^>]+m="([^"]+)"',page,re.I):
            raw=html.unescape(m.group(1)).replace('&quot;','"')
            try: meta=json.loads(raw)
            except Exception: continue
            u=meta.get("turl") or meta.get("murl")
            title=meta.get("t") or meta.get("purl") or "Bing image"
            if valid_url(u): candidates.append((relevance(title),u,title,"Bing Images"))
        # Fallback parser for changed Bing markup.
        for m in re.finditer(r'"murl":"(https?[^"\\]+)"',page):
            u=m.group(1).replace('\\/','/')
            if valid_url(u): candidates.append((relevance(search_query),u,"Bing result","Bing Images"))
    except Exception as e:
        print("[image] Bing:",e)

    # 3) Google Images thumbnail fallback. This is useful for models absent from Commons.
    try:
        google="https://www.google.com/search?tbm=isch&q="+quote(search_query)
        req=Request(google,headers={"User-Agent":"Mozilla/5.0 (X11; Linux x86_64) AppleWebKit/537.36 Chrome/131 Safari/537.36"})
        with urlopen(req,timeout=12) as r:
            page=r.read().decode("utf-8","ignore")
        patterns=[
            r'https://encrypted-tbn0\.gstatic\.com/images\?[^"\\ ]+',
            r'https://encrypted-tbn[0-9]+\.gstatic\.com/images\?[^"\\ ]+'
        ]
        seen=set()
        for pat in patterns:
            for m in re.finditer(pat,page):
                u=html.unescape(m.group(0)).replace('\\u003d','=').replace('\\u0026','&')
                if u not in seen and valid_url(u):
                    seen.add(u); candidates.append((max(3,relevance(search_query)),u,"Google Images thumbnail","Google Images"))
                if len(seen)>=12: break
    except Exception as e:
        print("[image] Google:",e)

    # Highest relevance first. Prefer sources that provide a direct image URL.
    candidates.sort(key=lambda x:x[0],reverse=True)
    seen=set()
    for score,u,title,source in candidates:
        if u in seen: continue
        seen.add(u)
        # Do not return obviously unrelated search hits.
        if score >= 8 or (source == "Google Images" and model_tokens):
            return {"ok":True,"image":u,"title":search_query,"source":source,"matched_file":title}

    # Final visual fallback: use a Google image-search thumbnail if available,
    # even when metadata was poor. This keeps the GUI visual instead of NO IMAGE FOUND.
    if candidates:
        score,u,title,source=candidates[0]
        return {"ok":True,"image":u,"title":search_query,"source":source,"matched_file":title,"fallback":True}

    return {"ok":False,"message":"Image service unavailable","title":search_query}

function label(k){return k.replace(/_/g," ").replace(/\b\w/g,c=>c.toUpperCase())}
function normKey(k){return String(k).toLowerCase().replace(/[^a-z0-9]/g,"")}
function get(d,keys){
 const map={};
 for(const [k,v] of Object.entries(d||{})) map[normKey(k)]=v;
 for(const k of keys){const nk=normKey(k);if(map[nk]!==undefined&&map[nk]!==null&&map[nk]!==""&&map[nk]!=="N/A")return map[nk]}
 return "N/A";
}

async function loadVehicleImage(d, rc) {
    let make = get(d, ["maker", "manufacturer", "make", "model_name"]);
    let model = get(d, ["model", "maker_model", "maker model"]);
    
    let cleanMake = make;
    if (cleanMake !== "N/A" && cleanMake !== "Unknown") {
        cleanMake = cleanMake.replace(/\s*\([^)]*\)\s*/g, " ").replace(/\s*P\.?\s*LTD\.?\s*/g, " ").trim();
    }
    
    // IMPORTANT: search with BOTH manufacturer + model.
    // Searching only "MAESTRO" is ambiguous and can return no/unrelated images.
    let query = "";
    if (model && model !== "N/A" && model !== "Unknown") {
        if (cleanMake && cleanMake !== "N/A" && cleanMake !== "Unknown") {
            query = cleanMake + " " + model + " vehicle";
        } else {
            query = model + " vehicle";
        }
    } else if (cleanMake && cleanMake !== "N/A" && cleanMake !== "Unknown") {
        query = cleanMake + " vehicle";
    } else {
        query = rc;
    }
    
    log("SEARCH QUERY: " + query);
    
    let displayModel = model && model !== "N/A" && model !== "Unknown" ? model : (cleanMake || rc);
    
    const imgElement = document.getElementById('vehicleImg');
    const placeholderElement = document.getElementById('placeholder');
    const modelNameElement = document.getElementById('modelName');
    
    modelNameElement.textContent = "MODEL: " + displayModel.toUpperCase();
    
    lastImageQuery = query;
    
    // ===== IMAGE VISIBILITY FIX =====
    imgElement.className = "vehicle-img hidden";
    imgElement.style.display = "none";
    placeholderElement.className = "placeholder visible";
    placeholderElement.style.display = "block";
    document.getElementById('imageSource').textContent = "";
    document.getElementById('imageStatus').textContent = "";
    
    if (!query || query.trim() === "") {
        placeholderElement.innerHTML = '<div class="car">♧</div>NO MODEL DATA<br><br>⚠️ Vehicle model information not available';
        log("No vehicle model data available for image search");
        return;
    }
    
    try {
        log("Searching image for: " + query);
        document.getElementById('imageStatus').textContent = "⏳ Searching for image...";
        document.getElementById('imageStatus').style.color = "#b6a66f";
        const imageUrl = "/api/vehicle-image?q=" + encodeURIComponent(query) + "&make=" + encodeURIComponent(make || "") + "&model=" + encodeURIComponent(model || "");
        const r = await fetch(imageUrl, {cache:"no-store"});
        const x = await r.json();
        if (x.ok && x.image) {
            // ===== SHOW IMAGE - FORCE VISIBLE =====
            const directImage = x.image;
            const proxiedImage = "/api/image-proxy?url=" + encodeURIComponent(directImage);
            imgElement.onerror = function(){
                if (this.dataset.directTried !== "1") {
                    this.dataset.directTried = "1";
                    this.src = directImage;
                }
            };
            imgElement.dataset.directTried = "0";
            imgElement.src = proxiedImage;
            imgElement.className = "vehicle-img visible";
            imgElement.style.display = "block";
            imgElement.style.maxWidth = "95%";
            imgElement.style.maxHeight = "90%";
            imgElement.style.objectFit = "contain";
            imgElement.style.margin = "auto";
            imgElement.style.borderRadius = "8px";
            
            // ===== HIDE PLACEHOLDER =====
            placeholderElement.className = "placeholder hidden";
            placeholderElement.style.display = "none";
            
            // Always show the actual vehicle model from the RTO data.
            // Do not replace it with a Wikimedia filename.
            modelNameElement.textContent = "MODEL: " + displayModel.toUpperCase();
            
            if (x.source) {
                document.getElementById('imageSource').textContent = "📷 Source: " + x.source;
                lastImageSource = x.source;
            }
            document.getElementById('imageStatus').textContent = "✅ Image loaded successfully!";
            document.getElementById('imageStatus').style.color = "#2dff83";
            log("Model image loaded: " + query + " from " + (x.source || "unknown source") +
                (x.matched_file ? " | matched: " + x.matched_file : ""));
        } else {
            // Never leave the panel on a "NO IMAGE FOUND" state.
            // Give the user a direct image-search fallback instead.
            const fallbackQ = encodeURIComponent((make ? make + " " : "") + (model || query) + " vehicle");
            const fallbackUrl = "https://www.google.com/search?tbm=isch&q=" + fallbackQ;
            placeholderElement.className = "placeholder visible";
            placeholderElement.style.display = "block";
            placeholderElement.innerHTML = '<div class="car">🔎</div><b>IMAGE SEARCH</b><br><br>' +
                'No direct thumbnail was returned.<br><br>' +
                '<a href="' + fallbackUrl + '" target="_blank" rel="noopener" style="color:#2dff83">Open vehicle images</a>';
            imgElement.className = "vehicle-img hidden";
            imgElement.style.display = "none";
            document.getElementById('imageSource').textContent = "🔎 Image search fallback";
            document.getElementById('imageStatus').textContent = "Image search ready";
            document.getElementById('imageStatus').style.color = "#2dff83";
        }
    } catch (e) {
        log("Image lookup error: " + e.message);
        const fallbackQ = encodeURIComponent((make ? make + " " : "") + (model || query) + " vehicle");
        const fallbackUrl = "https://www.google.com/search?tbm=isch&q=" + fallbackQ;
        placeholderElement.className = "placeholder visible";
        placeholderElement.style.display = "block";
        placeholderElement.innerHTML = '<div class="car">🔎</div><b>VEHICLE IMAGE SEARCH</b><br><br>' +
            '<a href="' + fallbackUrl + '" target="_blank" rel="noopener" style="color:#2dff83">Open images for ' +
            (model || query).replace(/</g,"&lt;") + '</a>';
        imgElement.className = "vehicle-img hidden";
        imgElement.style.display = "none";
        document.getElementById('imageSource').textContent = "🔎 Search fallback";
        document.getElementById('imageStatus').textContent = "Image search available";
        document.getElementById('imageStatus').style.color = "#2dff83";
    }
}

ALLOWED_IMAGE_HOSTS = {
    "upload.wikimedia.org", "commons.wikimedia.org",
    "encrypted-tbn0.gstatic.com", "encrypted-tbn1.gstatic.com",
    "encrypted-tbn2.gstatic.com", "encrypted-tbn3.gstatic.com",
}

def allowed_image_host(host):
    host=(host or "").lower().split(":")[0]
    return host in ALLOWED_IMAGE_HOSTS or (host.endswith(".mm.bing.net") and host.startswith("tse"))

def fetch_image_bytes(image_url):
    """Download only from known public image/CDN hosts and return bytes + content type."""
    p=urlparse(image_url)
    if p.scheme not in ("http", "https") or not allowed_image_host(p.netloc):
        return None, None
    req=Request(image_url,headers={
        "User-Agent":"Mozilla/5.0 (X11; Linux x86_64) AppleWebKit/537.36 Chrome/131 Safari/537.36",
        "Accept":"image/avif,image/webp,image/apng,image/svg+xml,image/*,*/*;q=0.8",
    })
    with urlopen(req,timeout=15) as r:
        data=r.read(8*1024*1024+1)
        if len(data)>8*1024*1024:
            return None,None
        ctype=(r.headers.get("Content-Type") or "image/jpeg").split(";")[0].strip().lower()
        if not ctype.startswith("image/"):
            return None,None
        return data,ctype
