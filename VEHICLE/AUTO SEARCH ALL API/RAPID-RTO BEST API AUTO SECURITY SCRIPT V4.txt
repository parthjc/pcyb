class SecureLicenseManager:
    """Enhanced secure license management system"""
    
    def __init__(self):
        self.license_file = "license.lic"
        self.version = "9.6.0 AUTO-FILTER"
        self.author = "ZYNTRASEC Security"
        self.copyright = "© 2026 ZYNTRASEC Security. All Rights Reserved."
        self.year = "2026"
        self.expiry_date = "2030-12-31"
        self.master_key = self._get_master_key()
        self.license_key = self._get_secure_license_key()
    
    def _get_mac_address(self) -> Optional[str]:
        """Get MAC address securely across platforms"""
        try:
            if platform.system() == "Windows":
                try:
                    import winreg
                    key = winreg.OpenKey(winreg.HKEY_LOCAL_MACHINE, 
                                       r"SYSTEM\CurrentControlSet\Control\Class\{4d36e972-e325-11ce-bfc1-08002be10318}")
                    for i in range(100):
                        try:
                            subkey = winreg.EnumKey(key, i)
                            skey = winreg.OpenKey(key, subkey)
                            mac = winreg.QueryValueEx(skey, "MACAddress")[0]
                            if mac and mac != "00:00:00:00:00:00":
                                return mac
                        except:
                            continue
                except:
                    pass
            
            elif platform.system() == "Linux":
                try:
                    import fcntl
                    import struct
                    s = socket.socket(socket.AF_INET, socket.SOCK_DGRAM)
                    info = fcntl.ioctl(s.fileno(), 0x8927, struct.pack('256s', b'eth0'))
                    return ':'.join('%02x' % b for b in info[18:24])
                except:
                    pass
            
            elif platform.system() == "Darwin":  # macOS
                try:
                    import fcntl
                    import struct
                    s = socket.socket(socket.AF_INET, socket.SOCK_DGRAM)
                    info = fcntl.ioctl(s.fileno(), 0x8927, struct.pack('256s', b'en0'))
                    return ':'.join('%02x' % b for b in info[18:24])
                except:
                    pass
            
            # Fallback methods
            try:
                import uuid
                mac = uuid.getnode()
                return ':'.join(['{:02x}'.format((mac >> elements) & 0xff) for elements in range(5, -1, -1)])
            except:
                pass
            
            try:
                result = subprocess.check_output(['ifconfig'], text=True)
                mac_match = re.search(r'ether\s+([0-9a-fA-F:]{17})', result)
                if mac_match:
                    return mac_match.group(1)
            except:
                pass
            
            try:
                result = subprocess.check_output(['ip', 'link'], text=True)
                mac_match = re.search(r'link/ether\s+([0-9a-fA-F:]{17})', result)
                if mac_match:
                    return mac_match.group(1)
            except:
                pass
            
            return None
            
        except Exception as e:
            print(f"⚠️ MAC detection error: {e}")
            return self._get_fallback_mac()
    
    def _get_fallback_mac(self) -> str:
        """Fallback MAC generation"""
        return ':'.join(['{:02x}'.format((uuid.getnode() >> elements) & 0xff) 
                        for elements in range(5, -1, -1)])
    
    def _get_master_key(self) -> bytes:
        """Generate a machine-specific master key"""
        try:
            identifiers = [
                self._get_mac_address() or "00:00:00:00:00:00",
                platform.node() or "unknown",
                platform.system() or "unknown",
                platform.machine() or "unknown",
                str(uuid.getnode()),
                os.path.realpath(sys.executable) if hasattr(sys, 'executable') else "python"
            ]
            
            combined = "|".join(identifiers).encode()
            kdf = PBKDF2HMAC(
                algorithm=hashes.SHA256(),
                length=32,
                salt=b"ZYNTRASEC_SALT_2026_V2",
                iterations=100000,
            )
            return base64.urlsafe_b64encode(kdf.derive(combined))
        except Exception as e:
            print(f"⚠️ Master key generation error: {e}")
            return base64.urlsafe_b64encode(b"ZYNTRASEC_MASTER_KEY_2026")
    
    def _get_secure_license_key(self) -> str:
        """Retrieve or generate secure license key"""
        env_key = os.environ.get('ZYNTRASEC_LICENSE_KEY')
        if env_key:
            return env_key
        
        if os.path.exists(self.license_file):
            try:
                with open(self.license_file, 'r') as f:
                    data = json.load(f)
                if 'data' in data:
                    decrypted = self._decrypt_data(data['data'])
                    if decrypted and 'key' in decrypted:
                        return decrypted['key']
            except:
                pass
        
        return self._generate_license_key()
    
    def _generate_license_key(self) -> str:
        """Generate a cryptographically secure license key"""
        import secrets
        import string
        
        alphabet = string.ascii_letters + string.digits + "!@#$%^&*"
        key = ''.join(secrets.choice(alphabet) for _ in range(64))
        
        mac = self._get_mac_address() or "000000000000"
        signature = hashlib.sha256(f"{key}{mac}".encode()).hexdigest()[:16]
        
        return f"{key}-{signature}"
    
    def _encrypt_data(self, data: Dict) -> str:
        """Encrypt license data"""
        try:
            fernet = Fernet(self.master_key)
            json_data = json.dumps(data).encode()
            encrypted = fernet.encrypt(json_data)
            return base64.urlsafe_b64encode(encrypted).decode()
        except Exception as e:
            print(f"❌ Encryption error: {e}")
            return ""
    
    def _decrypt_data(self, encrypted_data: str) -> Dict:
        """Decrypt license data"""
        try:
            fernet = Fernet(self.master_key)
            encrypted_bytes = base64.urlsafe_b64decode(encrypted_data.encode())
            decrypted = fernet.decrypt(encrypted_bytes)
            return json.loads(decrypted.decode())
        except Exception as e:
            print(f"❌ Decryption error: {e}")
            return {}
    
    def verify_license(self) -> bool:
        """Verify license with multiple security checks"""
        try:
            if not os.path.exists(self.license_file):
                return self._handle_no_license()
            
            with open(self.license_file, 'r') as f:
                license_data = json.load(f)
            
            if 'key_hash' in license_data and 'data' not in license_data:
                print("⚠️ Old license format detected. Generating new secure license...")
                self.generate_license()
                return self.verify_license()
            
            if 'data' in license_data:
                decrypted = self._decrypt_data(license_data['data'])
                if not decrypted:
                    return False
                
                if not self._verify_license_components(decrypted):
                    return False
                
                return True
            
            return False
            
        except Exception as e:
            print(f"❌ License verification error: {e}")
            return False
    
    def _verify_license_components(self, data: Dict) -> bool:
        """Verify all license components"""
        # Check version
        if data.get('version') != self.version:
            print(f"❌ Version mismatch: {data.get('version')} != {self.version}")
            return False
        
        # Check expiry
        try:
            current = datetime.now().date()
            expiry = datetime.strptime(data.get('expiry', self.expiry_date), "%Y-%m-%d").date()
            if current > expiry:
                print(f"❌ License expired on {expiry}")
                return False
        except:
            print("❌ Invalid expiry date")
            return False
        
        # Check MAC address
        current_mac = self._get_mac_address()
        if current_mac and data.get('mac'):
            if current_mac.lower() != data['mac'].lower():
                print(f"❌ MAC address mismatch")
                return False
        
        # Check signature
        try:
            expected = hashlib.sha256(
                f"{data['key']}{data['mac']}{data['expiry']}{self.master_key.decode()}".encode()
            ).hexdigest()
            
            if data.get('signature') != expected:
                print(f"❌ Invalid license signature")
                return False
        except:
            return False
        
        return True
    
    def _handle_no_license(self) -> bool:
        """Handle case where no license file exists"""
        print(f"{Colors.YELLOW}  ⚠️ License not found.{Colors.END}")
        print(f"{Colors.CYAN}  🔒 Enter license key to continue{Colors.END}")
        print(f"{Colors.DIM}  💡 Or use --generate-license to create one{Colors.END}")
        entered = input(f"{Colors.MAGENTA}  🔑 License Key: {Colors.END}").strip()
        
        if self._validate_license_key(entered):
            print(f"{Colors.GREEN}  ✅ License verified!{Colors.END}")
            self.generate_license(entered)
            return True
        else:
            print(f"{Colors.RED}  ❌ Invalid license key!{Colors.END}")
            return False
    
    def _validate_license_key(self, key: str) -> bool:
        """Validate the entered license key"""
        if len(key) < 32:
            return False
        
        try:
            parts = key.split('-')
            if len(parts) != 2:
                return False
            
            key_part = parts[0]
            sig_part = parts[1]
            
            mac = self._get_mac_address() or "000000000000"
            expected_sig = hashlib.sha256(f"{key_part}{mac}".encode()).hexdigest()[:16]
            
            return sig_part == expected_sig
            
        except Exception:
            return False
    
    def generate_license(self, key: str = None) -> None:
        """Generate a secure license file"""
        mac = self._get_mac_address() or "00:00:00:00:00:00"
        
        if not key:
            key = self._generate_license_key()
        
        license_data = {
            'key': key,
            'mac': mac,
            'expiry': self.expiry_date,
            'version': self.version,
            'generated': datetime.now().isoformat()
        }
        
        license_data['signature'] = hashlib.sha256(
            f"{key}{mac}{self.expiry_date}{self.master_key.decode()}".encode()
        ).hexdigest()
        
        encrypted = self._encrypt_data(license_data)
        
        data_to_save = {
            'data': encrypted,
            'format': 'secure_v2',
            'timestamp': datetime.now().isoformat()
        }
        
        with open(self.license_file, 'w') as f:
            json.dump(data_to_save, f, indent=4)
        
        print(f"{Colors.GREEN}  ✅ Secure license file generated: {self.license_file}{Colors.END}")
        print(f"{Colors.DIM}     Key: {key}{Colors.END}")
        print(f"{Colors.DIM}     MAC: {mac}{Colors.END}")
        print(f"{Colors.DIM}     Expiry: {self.expiry_date}{Colors.END}")
        print(f"{Colors.DIM}     Version: {self.version}{Colors.END}")
    
    def get_license_info(self) -> Dict:
        """Get license information"""
        try:
            if not os.path.exists(self.license_file):
                return {'status': 'not_found'}
            
            with open(self.license_file, 'r') as f:
                data = json.load(f)
            
            if 'data' in data:
                decrypted = self._decrypt_data(data['data'])
                if decrypted:
                    return {
                        'status': 'valid',
                        'expiry': decrypted.get('expiry'),
                        'version': decrypted.get('version'),
                        'generated': decrypted.get('generated'),
                        'mac': decrypted.get('mac'),
                        'key_preview': decrypted.get('key', '')[:20] + '...'
                    }
            
            return {'status': 'invalid'}
        except Exception as e:
            return {'status': 'error', 'message': str(e)}

class SecureKeyManager:
    """Secure API key management with encryption"""
    
    def __init__(self):
        self.keys_file = "api_keys.json"
        self._init_storage()
    
    def _get_encryption_key(self) -> bytes:
        """Get encryption key for API keys"""
        try:
            system_info = f"{platform.node()}{platform.system()}{uuid.getnode()}{platform.machine()}"
            kdf = PBKDF2HMAC(
                algorithm=hashes.SHA256(),
                length=32,
                salt=b"ZYNTRASEC_API_SALT_2026",
                iterations=100000,
            )
            return base64.urlsafe_b64encode(kdf.derive(system_info.encode()))
        except:
            return base64.urlsafe_b64encode(b"ZYNTRASEC_API_KEY_2026")
    
    def _encrypt_value(self, value: str) -> str:
        """Encrypt a value"""
        if not value:
            return ""
        try:
            fernet = Fernet(self._get_encryption_key())
            return base64.urlsafe_b64encode(fernet.encrypt(value.encode())).decode()
        except:
            return ""
    
    def _decrypt_value(self, encrypted: str) -> str:
        """Decrypt a value"""
        if not encrypted:
            return ""
        try:
            fernet = Fernet(self._get_encryption_key())
            return fernet.decrypt(base64.urlsafe_b64decode(encrypted.encode())).decode()
        except:
            return ""
    
    def _init_storage(self):
        """Initialize secure key storage"""
        if not os.path.exists(self.keys_file):
            self._create_default_keys()
    
    def _create_default_keys(self):
        """Create default encrypted keys"""
        default_data = {
            'keys': {
                'key1': self._encrypt_value("cc54872d5dmshee8d02a0f0e71f6p14936bjsn5d397631d9c2"),
                'key2': "",
                'key3': ""
            },
            'active': 'key1',
            'key_names': {
                'key1': 'Primary Key',
                'key2': 'Backup Key 1',
                'key3': 'Backup Key 2'
            },
            'last_updated': datetime.now().isoformat(),
            'encrypted': True,
            'version': '2.0'
        }
        with open(self.keys_file, 'w') as f:
            json.dump(default_data, f, indent=4)
    
    def load_keys(self) -> Dict:
        """Load and decrypt keys"""
        try:
            with open(self.keys_file, 'r') as f:
                data = json.load(f)
            
            if data.get('encrypted', False):
                decrypted_keys = {}
                for key_id, encrypted_key in data.get('keys', {}).items():
                    decrypted_keys[key_id] = self._decrypt_value(encrypted_key)
                data['keys'] = decrypted_keys
            
            return data
        except Exception as e:
            print(f"⚠️ Error loading keys: {e}")
            return {'keys': {'key1': '', 'key2': '', 'key3': ''}, 'active': 'key1'}
    
    def save_keys(self, data: Dict) -> bool:
        """Save encrypted keys"""
        try:
            encrypted_keys = {}
            for key_id, key_value in data.get('keys', {}).items():
                encrypted_keys[key_id] = self._encrypt_value(key_value)
            
            data['keys'] = encrypted_keys
            data['encrypted'] = True
            data['last_updated'] = datetime.now().isoformat()
            data['version'] = '2.0'
            
            with open(self.keys_file, 'w') as f:
                json.dump(data, f, indent=4)
            return True
        except Exception as e:
            print(f"❌ Error saving keys: {e}")
            return False
    
    def get_active_key(self) -> str:
        """Get active key"""
        data = self.load_keys()
        active = data.get('active', 'key1')
        return data.get('keys', {}).get(active, "")
    
    def get_all_keys(self) -> List[tuple]:
        """Get all valid keys"""
        data = self.load_keys()
        keys = data.get('keys', {})
        valid_keys = []
        for key_id, key_value in keys.items():
            if key_value and key_value.strip():
                valid_keys.append((key_id, key_value))
        return valid_keys

class RateLimitTracker:
    def __init__(self):
        self.rate_limited_apis: Set[str] = set()
        self.rate_limited_urls: Set[str] = set()
        self.rate_limit_cooldown: Dict[str, float] = {}
        self.cooldown_seconds = 60
    
    def is_rate_limited(self, api_name: str, api_url: str) -> bool:
        if api_url in self.rate_limit_cooldown:
            if time.time() < self.rate_limit_cooldown[api_url]:
                return True
            else:
                self.remove_from_rate_limit(api_url)
                return False
        return api_name in self.rate_limited_apis or api_url in self.rate_limited_urls
    
    def mark_rate_limited(self, api_name: str, api_url: str):
        self.rate_limited_apis.add(api_name)
        self.rate_limited_urls.add(api_url)
        self.rate_limit_cooldown[api_url] = time.time() + self.cooldown_seconds
        print(f"  {Colors.YELLOW}🚫 Auto-skipping: {api_name} (Rate Limited - {self.cooldown_seconds}s cooldown){Colors.END}")
    
    def remove_from_rate_limit(self, api_url: str):
        to_remove = []
        for name in self.rate_limited_apis:
            for api in WORKING_APIS:
                if api.get('name') == name and api.get('url') == api_url:
                    to_remove.append(name)
                    break
        for name in to_remove:
            self.rate_limited_apis.discard(name)
        self.rate_limited_urls.discard(api_url)
        self.rate_limit_cooldown.pop(api_url, None)
    
    def reset_all(self):
        self.rate_limited_apis.clear()
        self.rate_limited_urls.clear()
        self.rate_limit_cooldown.clear()
    
    def get_limited_count(self) -> int:
        return len(self.rate_limited_urls)
    
    def get_limited_list(self) -> List[str]:
        return list(self.rate_limited_apis)

def should_retry(status_code: int, error_text: str = "") -> bool:
    retry_codes = RETRY_CONFIG["retry_status_codes"]
    if status_code in retry_codes:
        return True
    rate_limit_keywords = ['rate limit', 'too many', 'quota', 'exceeded', '429']
    if any(keyword in error_text.lower() for keyword in rate_limit_keywords):
        return True
    service_keywords = ['service unavailable', '503', 'gateway', 'timeout']
    if any(keyword in error_text.lower() for keyword in service_keywords):
        return True
    return False

def calculate_backoff(attempt: int) -> float:
    base_delay = RETRY_CONFIG["base_delay"]
    max_delay = RETRY_CONFIG["max_delay"]
    backoff_factor = RETRY_CONFIG["backoff_factor"]
    jitter = RETRY_CONFIG["jitter"]
    delay = min(base_delay * (backoff_factor ** attempt), max_delay)
    if jitter:
        jitter_amount = delay * 0.3
        delay = delay + random.uniform(-jitter_amount, jitter_amount)
    return max(0.5, delay)

def parse_rtosearch_html(html_text: str) -> Dict:
    try:
        soup = BeautifulSoup(html_text, 'html.parser')
        result = {}
        patterns = {
            'owner_name': ['Owner Name', 'Owner', 'Name'],
            'registration_no': ['Registration No', 'Vehicle No', 'RC Number'],
            'registration_date': ['Registration Date', 'Reg Date'],
            'vehicle_class': ['Vehicle Class', 'Class'],
            'maker': ['Maker', 'Manufacturer', 'Make'],
            'model': ['Model'],
            'fuel_type': ['Fuel Type', 'Fuel'],
            'engine_no': ['Engine No', 'Engine Number'],
            'chassis_no': ['Chassis No', 'Chassis Number'],
            'insurance_company': ['Insurance Company', 'Insurance'],
            'insurance_upto': ['Insurance Upto', 'Insurance Valid'],
            'color': ['Color'],
            'manufactured_year': ['Manufactured Year', 'Year']
        }
        
        text = soup.get_text()
        lines = [line.strip() for line in text.split('\n') if line.strip()]
        for i, line in enumerate(lines):
            for field, keywords in patterns.items():
                for keyword in keywords:
                    if keyword.lower() in line.lower():
                        if ':' in line:
                            parts = line.split(':', 1)
                            if len(parts) == 2:
                                result[field] = parts[1].strip()
                        elif i + 1 < len(lines):
                            result[field] = lines[i + 1].strip()
                        break
        
        tables = soup.find_all('table')
        for table in tables:
            rows = table.find_all('tr')
            for row in rows:
                cells = row.find_all(['td', 'th'])
                if len(cells) >= 2:
                    key = cells[0].get_text().strip()
                    value = cells[1].get_text().strip()
                    for field, keywords in patterns.items():
                        for keyword in keywords:
                            if keyword.lower() in key.lower():
                                result[field] = value
                                break
        return result
    except Exception as e:
        return {"error": f"Parse error: {str(e)}"}

def fetch_with_retry(rc: str, api_config: Dict, max_retries: int = None, is_auto_mode: bool = False) -> Dict:
    api_name = api_config.get("name", "Unknown")
    api_url = api_config.get("url", "")
    
    if is_auto_mode and is_excluded_from_auto_mode(api_config):
        print(f"  {Colors.RED}⏭️ Skipping: {api_name} (Excluded from AUTO MODE){Colors.END}")
        return {"error": "Excluded from AUTO MODE", "skipped": True}
    
    if max_retries is None:
        max_retries = RETRY_CONFIG["max_retries"]
    
    if is_auto_mode and RETRY_CONFIG["auto_skip_429"]:
        if rate_tracker.is_rate_limited(api_name, api_url):
            print(f"  {Colors.RED}⏭️ Skipping: {api_name} (Rate limited - auto-skip){Colors.END}")
            return {"error": "Rate limited - auto-skipped", "skipped": True}
    
    attempt = 0
    last_error = None
    original_retries = max_retries
    
    while attempt < max_retries:
        try:
            params = {}
            data = None
            json_data = None
            
            if api_config.get("params_mapping"):
                for param_key, param_value in api_config["params_mapping"].items():
                    if param_value is None:
                        params[param_key] = rc
                    else:
                        params[param_key] = param_value
            
            if api_config.get("is_form"):
                data = params
                params = {}
            
            if api_config["method"] == "POST" and not api_config.get("is_form"):
                json_data = params
                params = {}
            
            timeout = 15 + (attempt * 2)
            
            headers = api_config.get("headers", {}).copy()
            if 'User-Agent' not in headers:
                headers['User-Agent'] = 'Mozilla/5.0 (Windows NT 10.0; Win64; x64) AppleWebKit/537.36 (KHTML, like Gecko) Chrome/120.0.0.0 Safari/537.36'
            
            if api_config["method"] == "GET":
                response = requests.get(api_config["url"], params=params, headers=headers, timeout=timeout)
            elif api_config["method"] == "POST":
                if api_config.get("is_form"):
                    response = requests.post(api_config["url"], data=data, headers=headers, timeout=timeout)
                else:
                    response = requests.post(api_config["url"], json=json_data, headers=headers, timeout=timeout)
            else:
                return {"error": f"Unsupported method: {api_config['method']}"}
            
            if response.status_code == 503:
                if attempt >= RETRY_CONFIG["max_503_retries"]:
                    print(f"  {Colors.RED}❌ {api_name}: Service unavailable (503) - Max retries ({RETRY_CONFIG['max_503_retries']}) reached{Colors.END}")
                    return {"error": f"HTTP 503 - Service Unavailable (max retries)", "status_code": 503}
                print(f"  {Colors.YELLOW}⚠️ {api_name}: Service unavailable (503) - Retry {attempt+1}/{RETRY_CONFIG['max_503_retries']}{Colors.END}")
                delay = calculate_backoff(attempt)
                print(f"  {Colors.DIM}   Waiting {delay:.1f}s before retry...{Colors.END}")
                time.sleep(delay)
                attempt += 1
                continue
            
            if response.status_code == 429:
                if is_auto_mode and RETRY_CONFIG["auto_skip_429"]:
                    rate_tracker.mark_rate_limited(api_name, api_url)
                    return {"error": f"HTTP 429 - Rate Limited (Auto-skipped)", "skipped": True}
                print(f"  {Colors.YELLOW}⚠️ Rate limit (429) - {api_name}{Colors.END}")
                if RETRY_CONFIG["switch_key_on_429"]:
                    if rotate_api_key():
                        attempt += 1
                        continue
                delay = calculate_backoff(attempt)
                print(f"  {Colors.YELLOW}⏳ Waiting {delay:.1f}s before retry (attempt {attempt+1}/{max_retries}){Colors.END}")
                time.sleep(delay)
                attempt += 1
                continue
            
            if should_retry(response.status_code, response.text):
                if response.status_code in RETRY_CONFIG["retry_status_codes"]:
                    delay = calculate_backoff(attempt)
                    print(f"  {Colors.YELLOW}⏳ Retrying in {delay:.1f}s (attempt {attempt+1}/{max_retries}){Colors.END}")
                    time.sleep(delay)
                    attempt += 1
                    continue
                return {"error": f"HTTP {response.status_code}"}
            
            if response.status_code == 200:
                if rate_tracker.is_rate_limited(api_name, api_url):
                    rate_tracker.remove_from_rate_limit(api_url)
                    print(f"  {Colors.GREEN}✅ {api_name} recovered from rate limit{Colors.END}")
                try:
                    return response.json()
                except:
                    return {"raw_text": response.text, "status": "html"}
            
            return {"error": f"HTTP {response.status_code}"}
            
        except requests.exceptions.Timeout:
            print(f"  {Colors.YELLOW}⏰ Timeout - {api_name} (attempt {attempt+1}/{max_retries}){Colors.END}")
            delay = calculate_backoff(attempt)
            time.sleep(delay)
            attempt += 1
            last_error = "Timeout"
        except requests.exceptions.ConnectionError:
            print(f"  {Colors.YELLOW}🔌 Connection error - {api_name} (attempt {attempt+1}/{max_retries}){Colors.END}")
            delay = calculate_backoff(attempt)
            time.sleep(delay)
            attempt += 1
            last_error = "Connection Error"
        except Exception as e:
            error_msg = str(e)
            print(f"  {Colors.YELLOW}⚠️ Error: {error_msg} - {api_name} (attempt {attempt+1}/{max_retries}){Colors.END}")
            if "rate limit" in error_msg.lower() or "too many" in error_msg.lower():
                if is_auto_mode and RETRY_CONFIG["auto_skip_429"]:
                    rate_tracker.mark_rate_limited(api_name, api_url)
                    return {"error": f"Rate Limited - {error_msg}", "skipped": True}
                if RETRY_CONFIG["switch_key_on_429"]:
                    if rotate_api_key():
                        attempt += 1
                        continue
            delay = calculate_backoff(attempt)
            time.sleep(delay)
            attempt += 1
            last_error = error_msg
    
    return {"error": f"Max retries ({original_retries}) exceeded. Last error: {last_error}"}

def validate_indian_rc(rc: str) -> bool:
    state_codes = [
        "AP","AR","AS","BR","CG","GA","GJ","HR","HP","JK","JH",
        "KA","KL","MP","MH","MN","ML","MZ","NL","OR","PB","RJ",
        "SK","TN","TR","UP","WB","AN","CH","DN","DD","DL","LD",
        "PY","TS","TG","LA"
    ]
    pattern = r'^(' + '|'.join(state_codes) + r')[-]?\d{1,2}[-]?[A-Z]{1,2}[-]?\d{4}$'
    return re.match(pattern, rc) is not None

def flatten_dict(obj: Any, parent_key: str = '', sep: str = '_') -> Dict:
    items = []
    if isinstance(obj, dict):
        for k, v in obj.items():
            new_key = f"{parent_key}{sep}{k}" if parent_key else k
            if isinstance(v, (dict, list)):
                items.extend(flatten_dict(v, new_key, sep=sep).items())
            else:
                items.append((new_key, v))
    elif isinstance(obj, list):
        for i, v in enumerate(obj):
            new_key = f"{parent_key}{sep}{i}" if parent_key else str(i)
            if isinstance(v, (dict, list)):
                items.extend(flatten_dict(v, new_key, sep=sep).items())
            else:
                items.append((new_key, v))
    return dict(items)

def normalize_data(raw_data: Dict) -> Dict:
    flat = flatten_dict(raw_data)
    normalized = {}
    for key, value in flat.items():
        clean_key = key
        for prefix in ['data_', 'result_', 'response_', 'vehicle_', 'rc_']:
            if key.startswith(prefix):
                clean_key = key[len(prefix):]
                break
        normalized[clean_key] = value
    normalized['_raw'] = raw_data
    return normalized

def rc_lookup_with_auto_retry(rc: str, api_config: Dict, is_auto_mode: bool = False) -> Dict:
    valid_keys = get_all_valid_keys()
    if not valid_keys:
        return {"error": "No API keys configured!"}
    
    active_key_id = get_active_key_id()
    ordered_keys = []
    for key_id, key_value in valid_keys:
        if key_id == active_key_id:
            ordered_keys.insert(0, (key_id, key_value))
        else:
            ordered_keys.append((key_id, key_value))
    
    for key_index, (key_id, key_value) in enumerate(ordered_keys):
        global RAPIDAPI_KEY
        current_key = RAPIDAPI_KEY
        RAPIDAPI_KEY = key_value
        if 'headers' in api_config and 'x-rapidapi-key' in api_config['headers']:
            api_config['headers']['x-rapidapi-key'] = key_value
        key_name = get_key_name(key_id)
        print(f"  {Colors.CYAN}🔑 Trying key: {key_id} ({key_name}){Colors.END}")
        result = fetch_with_retry(rc, api_config, is_auto_mode=is_auto_mode)
        RAPIDAPI_KEY = current_key
        if 'headers' in api_config and 'x-rapidapi-key' in api_config['headers']:
            api_config['headers']['x-rapidapi-key'] = current_key
        if "error" not in result or "status" in result:
            if "skipped" not in result:
                result['_key_used'] = key_id
            return result
        error_msg = result.get("error", "Unknown error")
        if result.get("skipped", False):
            return result
        if any(keyword in error_msg.lower() for keyword in ['429', 'rate limit', 'too many', 'quota']):
            print(f"  {Colors.YELLOW}⚠️ Rate limit on {key_id} - Trying next key{Colors.END}")
            continue
        if any(keyword in error_msg.lower() for keyword in ['503', 'service unavailable']):
            print(f"  {Colors.YELLOW}⚠️ Service unavailable on {key_id} - Trying next key{Colors.END}")
            continue
        print(f"  {Colors.YELLOW}⚠️ {key_id} failed: {error_msg}{Colors.END}")
        continue
    
    return {"error": "All keys failed after retries"}

def test_api_key(key):
    try:
        test_url = "https://vahan-vehicle-details-api.p.rapidapi.com/rc_v2.php"
        test_params = {"registration_no": "GJ05VC1234"}
        headers = {
            "x-rapidapi-host": "vahan-vehicle-details-api.p.rapidapi.com",
            "x-rapidapi-key": key
        }
        response = requests.get(test_url, params=test_params, headers=headers, timeout=10)
        if response.status_code in [200, 404, 429]:
            return True, f"Status: {response.status_code}"
        return False, f"Status: {response.status_code}"
    except Exception as e:
        return False, str(e)

def test_all_keys():
    print(f"\n{Colors.CYAN}  🧪 TEST ALL KEYS{Colors.END}")
    print(f"  {'─'*50}")
    
    keys_data, active = get_all_keys_with_status()
    print(f"\n  🔄 Testing all keys...\n")
    results = []
    for key_id, info in keys_data.items():
        key_value = info['key']
        name = info['name']
        if key_value:
            masked = key_value[:8] + "..." + key_value[-4:] if len(key_value) > 12 else "****"
            print(f"  Testing {Colors.GREEN}{key_id}{Colors.END}: {masked} {Colors.DIM}({name}){Colors.END}")
            is_valid, message = test_api_key(key_value)
            if is_valid:
                print(f"    {Colors.GREEN}✅ {message}{Colors.END}")
                results.append((key_id, True, message))
            else:
                print(f"    {Colors.RED}❌ {message}{Colors.END}")
                results.append((key_id, False, message))
        else:
            print(f"  Testing {Colors.RED}{key_id}{Colors.END}: [EMPTY] {Colors.DIM}({name}) - Skipped{Colors.END}")
    
    print(f"\n  {Colors.CYAN}📊 SUMMARY:{Colors.END}")
    print(f"  {'─'*50}")
    valid_count = sum(1 for r in results if r[1])
    total_count = len(results)
    print(f"  ✅ Valid: {valid_count}/{total_count}")
    print(f"  ❌ Invalid: {total_count - valid_count}/{total_count}")
    if results:
        print(f"\n  {Colors.YELLOW}📋 Details:{Colors.END}")
        for key_id, is_valid, message in results:
            status = Colors.GREEN + "✅" + Colors.END if is_valid else Colors.RED + "❌" + Colors.END
            print(f"    {key_id}: {status} {message}")
    print(f"\n  {Colors.GREEN}✅ Testing complete!{Colors.END}")
    input(f"\n{Colors.DIM}Press Enter to continue...{Colors.END}")
