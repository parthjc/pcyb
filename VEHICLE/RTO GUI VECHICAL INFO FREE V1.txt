def valid_rc(rc):
 s=re.sub(r"[^A-Z0-9]","",rc.upper())
 return bool(re.match(r"^(AP|AR|AS|BR|CG|GA|GJ|HR|HP|JK|JH|KA|KL|MP|MH|MN|ML|MZ|NL|OD|OR|PB|RJ|SK|TN|TR|UP|WB|AN|CH|DN|DD|DL|LD|PY|TS|TG|LA)\d{1,2}[A-Z]{1,3}\d{4}$",s))

def flatten(x,p=""):
 out={}
 if isinstance(x,dict):
  for k,v in x.items():
   n=f"{p}_{k}" if p else str(k)
   if isinstance(v,(dict,list)):out.update(flatten(v,n))
   else:out[n]=v
 elif isinstance(x,list):
  for i,v in enumerate(x):
   n=f"{p}_{i}" if p else str(i)
   if isinstance(v,(dict,list)):out.update(flatten(v,n))
   else:out[n]=v
 return out

def is_sensitive(k):
 k=str(k).lower().replace("-","_")
 return any(w in k for w in SENSITIVE)

def safe_result(raw):
 out={}
 for k,v in flatten(raw).items():
  out[k.lower()]="[MASKED — sensitive field]" if is_sensitive(k) else v
 return out

def api_get(api,rc):
 headers=dict(api["headers"])
 if api["name"].startswith("RTO Vehicle Info"):
  if not RAPIDAPI_KEY:return {"error":"RapidAPI key not configured"}
  headers["x-rapidapi-key"]=RAPIDAPI_KEY
 url=api["url"]+"?"+api["param"]+"="+quote(rc)
 try:
  with urlopen(Request(url,headers=headers,method="GET"),timeout=12) as r:
   if r.status!=200:return {"error":f"HTTP {r.status}"}
   body=r.read().decode("utf-8","replace")
   try:return json.loads(body)
   except:return {"error":"API returned non-JSON data"}
 except HTTPError as e:return {"error":f"HTTP {e.code}"}
 except URLError as e:return {"error":f"Connection error: {e.reason}"}
 except TimeoutError:return {"error":"Timeout"}
 except Exception as e:return {"error":str(e)}

def vehicle_image(query):
    if not query:
        return {"ok": False, "message": "No search query provided"}

    query = str(query).strip()
    query_lower = query.lower()
    
    UNICORN_IMAGE = "https://upload.wikimedia.org/wikipedia/commons/thumb/6/68/Honda_Unicorn_160.jpg/640px-Honda_Unicorn_160.jpg"

    # Fallback
    try:
        url = f"https://en.wikipedia.org/w/api.php?action=query&format=json&prop=pageimages&titles={quote(query)}&pithumbsize=400"
        req = Request(url, headers={"User-Agent": "ZYNTRASEC-Vehicle-GUI/2.0"})
        with urlopen(req, timeout=8) as r:
            data = json.loads(r.read().decode("utf-8", "replace"))
        for page in data.get("query", {}).get("pages", {}).values():
            thumb = page.get("thumbnail", {})
            src = thumb.get("source")
            if src:
                return {"ok": True, "image": src, "title": page.get("title", ""), "source": "Wikipedia"}
    except:
        pass
    
    return {"ok": False, "message": f"No image found for: {query}"}

function esc(x){return String(x??"").replace(/[&<>"']/g,m=>({"&":"&amp;","<":"&lt;",">":"&gt;",'"':"&quot;","'":"&#039;"}[m]))}

function label(k){return k.replace(/_/g," ").replace(/\b\w/g,c=>c.toUpperCase())}
function normKey(k){return String(k).toLowerCase().replace(/[^a-z0-9]/g,"")}
function get(d,keys){
 const map={};
 for(const [k,v] of Object.entries(d||{})) map[normKey(k)]=v;
 for(const k of keys){const nk=normKey(k);if(map[nk]!==undefined&&map[nk]!==null&&map[nk]!==""&&map[nk]!=="N/A")return map[nk]}
 return "N/A";
}

function buildSummary(d,rc){

 let rto=get(d,["rto","rto_office","registered_rto"]);
 return `${rc} • ${make} ${model} • Fuel: ${fuel} • RTO: ${rto}`.replace(/N\/A/g,"Unknown");
}

       let day = parseInt(parts[0]);
       if (day > 31) { day = 30; regDate = `${day.toString().padStart(2, '0')}-${parts[1]}-${parts[2]}`; }
     }
   } catch(e) {}
 }
 
 let maker = v(["maker","manufacturer","make","model_name"]);

   maker = maker.replace(/\s*\([^)]*\)\s*/g, " ").replace(/\s*P\.?\s*LTD\.?\s*/g, " ").trim();

async function loadVehicleImage(d, rc) {
    let make = get(d, ["maker", "manufacturer", "make", "model_name"]);
    let model = get(d, ["model", "maker_model", "maker model"]);
    
    let cleanMake = make;
    if (cleanMake !== "N/A" && cleanMake !== "Unknown") {
        cleanMake = cleanMake.replace(/\s*\([^)]*\)\s*/g, " ").replace(/\s*P\.?\s*LTD\.?\s*/g, " ").trim();
    }
    
    // ===== BUILD SEARCH QUERY - ALWAYS USE MODEL NAME =====
    let query = "";
    
    // FIRST: Use model name (MOST IMPORTANT)
    if (model && model !== "N/A" && model !== "Unknown") {
        if (model.toUpperCase().includes("UNICORN")) {
            query = "Honda Unicorn";
        } else if (model.toUpperCase().includes("ACTIVA")) {
            query = "Honda Activa";
        } else if (model.toUpperCase().includes("SHINE")) {
            query = "Honda Shine";
        } else if (model.toUpperCase().includes("PULSAR")) {
            query = "Bajaj Pulsar";
        } else if (model.toUpperCase().includes("APACHE")) {
            query = "TVS Apache";
        } else if (model.toUpperCase().includes("SPLENDOR")) {
            query = "Hero Splendor";
        } else if (model.toUpperCase().includes("CLASSIC") || model.toUpperCase().includes("BULLET")) {
            query = "Royal Enfield Classic";
        } else if (model.toUpperCase().includes("MAESTRO")) {
            query = "Hero Maestro";
        } else if (model.toUpperCase().includes("JUPITER")) {
            query = "TVS Jupiter";
        } else if (model.toUpperCase().includes("NTORQ")) {
            query = "TVS Ntorq";
        } else if (model.toUpperCase().includes("FZ")) {
            query = "Yamaha FZ";
        } else if (model.toUpperCase().includes("R15")) {
            query = "Yamaha R15";
        } else if (model.toUpperCase().includes("ACCESS")) {
            query = "Suzuki Access";
        } else if (model.toUpperCase().includes("OLA")) {
            query = "OLA S1 Pro";
        } else if (model.toUpperCase().includes("ATHER")) {
            query = "Ather 450X";
        } else {
            query = model;
        }
    }
    // SECOND: Use make name
    else if (cleanMake && cleanMake !== "N/A" && cleanMake !== "Unknown") {
        if (cleanMake.toUpperCase().includes("HONDA")) {
            query = "Honda Motorcycle";
        } else if (cleanMake.toUpperCase().includes("HERO")) {
            query = "Hero MotoCorp";
        } else if (cleanMake.toUpperCase().includes("TVS")) {
            query = "TVS Motor";
        } else if (cleanMake.toUpperCase().includes("BAJAJ")) {
            query = "Bajaj Auto";
        } else {
            query = cleanMake;
        }
    }
    // LAST RESORT: Use RC number
    else {
        query = rc;
    }
    
    log("SEARCH QUERY: " + query);  // Debug log
    
    // Display model name
    let displayModel = model && model !== "N/A" && model !== "Unknown" ? model : (cleanMake || rc);
    
    // ===== IMAGE DISPLAY ELEMENTS =====
    const imgElement = document.getElementById('vehicleImg');
    const placeholderElement = document.getElementById('placeholder');
    const modelNameElement = document.getElementById('modelName');
    
    // Update model name
    modelNameElement.textContent = "MODEL: " + displayModel.toUpperCase();
    
    lastImageQuery = query;
    
    // Hide image, show placeholder initially
    imgElement.className = "vehicle-img hidden";
    imgElement.style.display = "none";
    placeholderElement.className = "placeholder";
    placeholderElement.style.display = "block";
    document.getElementById('imageSource').textContent = "";
    document.getElementById('imageStatus').textContent = "";
    
    if (!query || query.trim() === "") {
        placeholderElement.innerHTML = '<div class="car">♧</div>NO MODEL DATA<br><br>⚠️ Vehicle model information not available';
        log("No vehicle model data available for image search");
        return;
    }
    
    try {
        log("Searching image for: " + query);
        document.getElementById('imageStatus').textContent = "⏳ Searching for image...";
        document.getElementById('imageStatus').style.color = "#b6a66f";
        const r = await fetch("/api/vehicle-image?q=" + encodeURIComponent(query));
        const x = await r.json();
        if (x.ok && x.image) {
            imgElement.src = x.image;
            imgElement.className = "vehicle-img";
            imgElement.style.display = "block";
            imgElement.style.maxWidth = "95%";
            imgElement.style.maxHeight = "90%";
            imgElement.style.objectFit = "contain";
            imgElement.style.margin = "auto";
            imgElement.style.borderRadius = "8px";
            
            placeholderElement.className = "placeholder hidden";
            placeholderElement.style.display = "none";
            
            let displayTitle = x.title || displayModel;
            modelNameElement.textContent = "MODEL: " + displayTitle.toUpperCase();
            
            if (x.source) {
                document.getElementById('imageSource').textContent = "📷 Source: " + x.source;
                lastImageSource = x.source;
            }
            document.getElementById('imageStatus').textContent = "✅ Image loaded successfully!";
            document.getElementById('imageStatus').style.color = "#2dff83";
            log("Model image loaded: " + query + " from " + (x.source || "unknown source"));
        } else {
            log("No image found for: " + query);
            placeholderElement.className = "placeholder";
            placeholderElement.style.display = "block";
            placeholderElement.innerHTML = '<div class="car">♧</div>NO IMAGE FOUND<br><br>📷 ' + query + '<br><br>Try clicking "REFRESH IMAGE" or "TRY ALTERNATE"<br><br>🔄 Searching with: ' + query;
            imgElement.className = "vehicle-img hidden";
            imgElement.style.display = "none";
            document.getElementById('imageSource').textContent = "❌ No image available";
            document.getElementById('imageStatus').textContent = "❌ No image found for: " + query;
            document.getElementById('imageStatus').style.color = "#ff5365";
        }
    } catch (e) {
        log("Image lookup error: " + e.message);
        placeholderElement.className = "placeholder";
        placeholderElement.style.display = "block";
        placeholderElement.innerHTML = '<div class="car">♧</div>IMAGE SERVICE ERROR<br><br>⚠️ ' + e.message;
        imgElement.className = "vehicle-img hidden";
        imgElement.style.display = "none";
        document.getElementById('imageSource').textContent = "⚠️ Error: " + e.message;
        document.getElementById('imageStatus').textContent = "⚠️ Error: " + e.message;
        document.getElementById('imageStatus').style.color = "#b6a66f";
    }
}

class Handler(BaseHTTPRequestHandler):

 def send_json(self,d,status=200):
  b=json.dumps(d,ensure_ascii=False).encode()
  self.send_response(status);self.send_header("Content-Type","application/json; charset=utf-8");self.send_header("Content-Length",str(len(b)));self.send_header("Cache-Control","no-store");self.end_headers();self.wfile.write(b)
 def do_GET(self):
  u=urlparse(self.path)
  if u.path=="/":
   b=HTML.encode();self.send_response(200);self.send_header("Content-Type","text/html; charset=utf-8");self.send_header("Content-Length",str(len(b)));self.end_headers();self.wfile.write(b);return
  if u.path=="/api/status":self.send_json({"online":True,"configured":bool(RAPIDAPI_KEY),"sources":len(APIS)});return
  if u.path=="/api/history":self.send_json(history());return
  if u.path=="/api/favorites":self.send_json(favorites());return
  if u.path=="/api/vehicle-image":
   q=parse_qs(u.query).get("q",[""])[0].strip();self.send_json(vehicle_image(q) if q else {"ok":False});return
  if u.path=="/api/lookup":
   rc=parse_qs(u.query).get("rc",[""])[0].strip().upper()
   if not valid_rc(rc):self.send_json({"ok":False,"error":"Invalid Indian RC format. Example: GJ05AB1234"},400);return
   d=lookup(rc);add_history(rc,d.get("api"),"SUCCESS" if d.get("ok") else "FAILED");self.send_json(d,200 if d.get("ok") else 502);return
  self.send_json({"error":"Not found"},404)
