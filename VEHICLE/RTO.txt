def validate_indian_rc(rc: str) -> bool:
    """Validate any Indian state RC (two-letter state code)"""
    # List of all Indian state/UT codes
    state_codes = [
        "AP", "AR", "AS", "BR", "CG", "GA", "GJ", "HR", "HP", "JK", "JH",
        "KA", "KL", "MP", "MH", "MN", "ML", "MZ", "NL", "OR", "PB", "RJ",
        "SK", "TN", "TR", "UP", "WB", "AN", "CH", "DN", "DD", "DL", "LD",
        "PY", "TS", "TG", "LA"
    ]
    # Regular expression: state code followed by digits, letters, digits
    pattern = r'^(' + '|'.join(state_codes) + r')[-]?\d{1,2}[-]?[A-Z]{1,2}[-]?\d{4}$'
    return re.match(pattern, rc) is not None

def get_state_from_rc(rc: str) -> str:
    """Extract state code from RC"""
    # Remove hyphens and get first two characters
    clean = re.sub(r'[-]', '', rc)
    return clean[:2].upper()

def fetch_from_rapidapi(rc: str) -> Dict:
    """Fetch data from RapidAPI"""
    try:
        params = {"registration_no": rc}
        response = requests.get(
            API["url"],
            params=params,
            headers=API["headers"],
            timeout=10
        )

        if response.status_code == 200:
            return response.json()
        else:
            return {"error": f"HTTP {response.status_code}"}

    except requests.exceptions.Timeout:
        return {"error": "Timeout"}
    except requests.exceptions.ConnectionError:
        return {"error": "Connection Error"}
    except Exception as e:
        return {"error": str(e)}

def fetch_from_backup(rc: str, api_config: Dict) -> Dict:
    """Fetch data from backup API"""
    try:
        params = {}
        if api_config.get("params"):
            for key, value in api_config["params"].items():
                params[key] = rc

        response = requests.get(
            api_config["url"],
            params=params,
            headers={},
            timeout=8
        )

        if response.status_code == 200:
            return response.json()
        return {"error": f"HTTP {response.status_code}"}

    except requests.exceptions.Timeout:
        return {"error": "Timeout"}
    except Exception as e:
        return {"error": str(e)}

def format_phone(phone: str) -> str:
    """Format phone number"""
    if not phone:
        return "N/A"
    phone = re.sub(r'[^\d+]', '', str(phone))
    if len(phone) == 10:
        return f"+91 {phone[:5]} {phone[5:]}"
    elif len(phone) == 11 and phone.startswith('0'):
        return f"+91 {phone[1:6]} {phone[6:]}"
    return phone if phone else "N/A"

        'road_tax_upto': '💰 Road Tax Upto',
        'registration_status': '📊 Registration Status',
        'color': '🎨 Color',
        'seating_capacity': '💺 Seating Capacity',
        'cubic_capacity': '🔧 Engine CC',
        'manufactured_year': '🏗️ Manufactured Year',
        'registered_state': '📍 Registered State'
    }

    # Clean phone if exists
    if 'phone' in data:
        data['phone'] = format_phone(data['phone'])
    if 'mobile' in data:
        data['mobile'] = format_phone(data['mobile'])

    # Display fields
    displayed = False
    for key, value in data.items():
        if key.startswith('_'):
            continue
        if key in unwanted:
            continue
        label = field_mapping.get(key, key.replace('_', ' ').title())
        display_value = value if value else 'N/A'
        print(f"  {Colors.YELLOW}{label:<30}{Colors.END} : {Colors.WHITE}{display_value}{Colors.END}")
        displayed = True

    if not displayed:
        print(f"{Colors.RED}  No vehicle data found for this RC number{Colors.END}")
        print(f"{Colors.DIM}  💡 Raw response: {json.dumps(data, indent=2)[:300]}...{Colors.END}")

    print(f"{Colors.GREEN}{'='*62}{Colors.END}")
    print(f"{Colors.CYAN}  📡 Data from: {api_name}{Colors.END}")
    print(f"{Colors.CYAN}  🏷️  State: {state}{Colors.END}")
    if is_demo:
        print(f"{Colors.YELLOW}  📌 Demo data for {rc} (Example){Colors.END}")
    print(f"{Colors.MAGENTA}  🔒 ZYNTRASEC SECURITY | Copyright {YEAR} | All Rights Reserved{Colors.END}")

def save_results(rc: str, data: Dict, is_demo: bool = False) -> str:
    """Save results to file"""
    timestamp = datetime.now().strftime("%Y%m%d_%H%M%S")
    prefix = "DEMO_" if is_demo else ""
    filename = f"ZYNTRASEC_{prefix}{rc}_{timestamp}.json"

    data['_metadata'] = {
        'tool': 'ZYNTRASEC Security Advanced',
        'version': VERSION,
        'year': YEAR,
        'timestamp': datetime.now().isoformat(),
        'copyright': COPYRIGHT,
        'rc_checked': rc,
        'is_demo': is_demo,
        'generated_by': 'ZYNTRASEC Security'
    }

    with open(filename, "w", encoding="utf-8") as f:
        json.dump(data, f, indent=4, ensure_ascii=False)

    print(f"\n  💾 Results saved to: {filename}")
    return filename
