def validate_indian_rc(rc: str) -> bool:
    state_codes = [
        "AP","AR","AS","BR","CG","GA","GJ","HR","HP","JK","JH",
        "KA","KL","MP","MH","MN","ML","MZ","NL","OR","PB","RJ",
        "SK","TN","TR","UP","WB","AN","CH","DN","DD","DL","LD",
        "PY","TS","TG","LA"
    ]
    pattern = r'^(' + '|'.join(state_codes) + r')[-]?\d{1,2}[-]?[A-Z]{1,2}[-]?\d{4}$'
    return re.match(pattern, rc) is not None

def fetch_from_api(rc: str, api_config: Dict) -> Dict:
    try:
        params = {}
        if api_config.get("params_mapping"):
            for param_key, _ in api_config["params_mapping"].items():
                params[param_key] = rc
        response = requests.get(
            api_config["url"],
            params=params,
            headers=api_config.get("headers", {}),
            timeout=10
        )
        if response.status_code == 200:
            return response.json()
        return {"error": f"HTTP {response.status_code}"}
    except Exception as e:
        return {"error": str(e)}

def flatten_dict(obj: Any, parent_key: str = '', sep: str = '_') -> Dict:
    items = []
    if isinstance(obj, dict):
        for k, v in obj.items():
            new_key = f"{parent_key}{sep}{k}" if parent_key else k
            if isinstance(v, (dict, list)):
                items.extend(flatten_dict(v, new_key, sep=sep).items())
            else:
                items.append((new_key, v))
    elif isinstance(obj, list):
        for i, v in enumerate(obj):
            new_key = f"{parent_key}{sep}{i}" if parent_key else str(i)
            if isinstance(v, (dict, list)):
                items.extend(flatten_dict(v, new_key, sep=sep).items())
            else:
                items.append((new_key, v))
    return dict(items)

def normalize_data(raw_data: Dict) -> Dict:
    flat = flatten_dict(raw_data)
    normalized = {}
    for key, value in flat.items():
        clean_key = key
        for prefix in ['data_', 'result_', 'response_', 'vehicle_', 'rc_']:
            if key.startswith(prefix):
                clean_key = key[len(prefix):]
                break
        normalized[clean_key] = value
    normalized['_raw'] = raw_data
    return normalized

async def handle_rc(update: Update, context: ContextTypes.DEFAULT_TYPE):
    """Process RC number sent by user."""
    rc = update.message.text.strip().upper()

    if not validate_indian_rc(rc):
        await update.message.reply_text(
            f"⚠️ *Invalid RC format*\n\n"
            f"Please enter a valid Indian RC number.\n"
            f"Example: `GJ19BC4041` or `GJ-19-BC-4041`\n\n"
            f"Try again.",
            parse_mode="Markdown",
            reply_markup=get_back_menu()
        )
        return

    # Send typing action
    await update.message.chat.send_action(action="typing")

    # Start searching
    status_msg = await update.message.reply_text(
        f"🔍 Searching for `{rc}`...\n\n"
        f"⏳ Checking 3 APIs...",
        parse_mode="Markdown"
    )

    data = None
    raw_data = None
    used_api = None
    attempts = []

    for api_config in WORKING_APIS:
        api_name = api_config["name"]
        try:
            result = fetch_from_api(rc, api_config)
            if "error" not in result and len(result) > 0:
                raw_data = result
                data = normalize_data(result)
                used_api = api_name
                break
            else:
                attempts.append(f"❌ {api_name}: {result.get('error', 'No data')}")
        except Exception as e:
            attempts.append(f"❌ {api_name}: {str(e)}")

    if data is not None:
        message_text = format_vehicle_info(data, rc, used_api)
        await status_msg.edit_text(message_text, reply_markup=get_back_menu())
    else:
        error_text = "\n".join(attempts) if attempts else "All APIs failed"
        await status_msg.edit_text(
            f"❌ *All APIs failed!*\n\n"
            f"Could not fetch data for `{rc}`.\n\n"
            f"*Attempts:*\n{error_text}\n\n"
            f"💡 *Suggestions:*\n"
            f"• Check internet connection\n"
            f"• Verify RC number\n"
            f"• Try a different RC",
            parse_mode="Markdown",
            reply_markup=get_back_menu()
        )
