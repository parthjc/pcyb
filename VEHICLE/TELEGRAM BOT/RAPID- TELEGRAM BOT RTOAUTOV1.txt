def get_bot_token():
    """Get bot token from various sources"""
    token_sources = [
        os.environ.get("TELEGRAM_BOT_TOKEN"),
        os.environ.get("BOT_TOKEN"),
    ]
    
    # Check for token file
    token_file = "bot_token.txt"
    if os.path.exists(token_file):
        try:
            with open(token_file, "r") as f:
                token = f.read().strip()
                if token:
                    token_sources.append(token)
        except:
            pass
    
    # Check config file
    config_file = "config.json"
    if os.path.exists(config_file):
        try:
            with open(config_file, "r") as f:
                config = json.load(f)
                token = config.get("bot_token")
                if token:
                    token_sources.append(token)
        except:
            pass
    
    # Return first valid token
    for token in token_sources:
        if token and token != "YOUR_TELEGRAM_BOT_TOKEN_HERE":
            return token
    
    return None

def validate_indian_rc(rc: str) -> bool:
    """Validate Indian RC number format"""
    state_codes = [
        "AP", "AR", "AS", "BR", "CG", "GA", "GJ", "HR", "HP", "JK", "JH",
        "KA", "KL", "MP", "MH", "MN", "ML", "MZ", "NL", "OR", "PB", "RJ",
        "SK", "TN", "TR", "UP", "WB", "AN", "CH", "DN", "DD", "DL", "LD",
        "PY", "TS", "TG", "LA"
    ]
    pattern = r'^(' + '|'.join(state_codes) + r')[-]?\d{1,2}[-]?[A-Z]{1,2}[-]?\d{4}$'
    return re.match(pattern, rc) is not None

def fetch_from_api(rc: str, api_config: Dict) -> Dict:
    """Fetch data from a single API"""
    try:
        params = {}
        if api_config.get("params_mapping"):
            for param_key, _ in api_config["params_mapping"].items():
                params[param_key] = rc

        response = requests.get(
            api_config["url"],
            params=params,
            headers=api_config.get("headers", {}),
            timeout=10
        )

        if response.status_code == 200:
            return response.json()
        else:
            return {"error": f"HTTP {response.status_code}"}
    except requests.exceptions.Timeout:
        return {"error": "Timeout"}
    except requests.exceptions.ConnectionError:
        return {"error": "Connection Error"}
    except Exception as e:
        return {"error": str(e)}

def flatten_dict(obj: Any, parent_key: str = '', sep: str = '_') -> Dict:
    """Flatten nested dictionary"""
    items = []
    if isinstance(obj, dict):
        for k, v in obj.items():
            new_key = f"{parent_key}{sep}{k}" if parent_key else k
            if isinstance(v, (dict, list)):
                items.extend(flatten_dict(v, new_key, sep=sep).items())
            else:
                items.append((new_key, v))
    elif isinstance(obj, list):
        for i, v in enumerate(obj):
            new_key = f"{parent_key}{sep}{i}" if parent_key else str(i)
            if isinstance(v, (dict, list)):
                items.extend(flatten_dict(v, new_key, sep=sep).items())
            else:
                items.append((new_key, v))
    return dict(items)

def normalize_data(raw_data: Dict) -> Dict:
    """Normalize API response data"""
    flat = flatten_dict(raw_data)
    normalized = {}
    for key, value in flat.items():
        clean_key = key
        for prefix in ['data_', 'result_', 'response_', 'vehicle_', 'rc_']:
            if key.startswith(prefix):
                clean_key = key[len(prefix):]
                break
        normalized[clean_key] = value
    normalized['_raw'] = raw_data
    return normalized

async def handle_rc_lookup(update: Update, context: ContextTypes.DEFAULT_TYPE):
    """Handle RC number lookup"""
    rc = update.message.text.strip().upper()
    user_id = update.effective_user.id

    # Track user
    if 'users' not in context.bot_data:
        context.bot_data['users'] = set()
    context.bot_data['users'].add(user_id)

    # Validate RC format
    if not validate_indian_rc(rc):
        await update.message.reply_text(
            f"⚠️ *Invalid RC Number Format*\n\n"
            f"`{rc}` doesn't appear to be a valid Indian RC number.\n\n"
            f"*Valid formats:*\n"
            f"• `GJ-05-VC-1234`\n"
            f"• `GJ05VC1234`\n"
            f"• `MH-12-AB-5678`\n\n"
            f"*State codes:* AP, GJ, MH, KA, TN, etc.\n\n"
            f"Send /help for more information.",
            parse_mode='Markdown'
        )
        return

    # Send searching message
    searching_msg = await update.message.reply_text(
        f"🔍 *Searching for RC:* `{rc}`\n"
        f"⏳ Please wait...",
        parse_mode='Markdown'
    )

    # Try all APIs
    data = None
    raw_data = None
    used_api = None
    attempted = 0
    errors = []

    for api_config in WORKING_APIS:
        attempted += 1
        api_name = api_config["name"]
        try:
            result = fetch_from_api(rc, api_config)

            if "error" not in result and len(result) > 0:
                raw_data = result
                data = normalize_data(result)
                used_api = api_name
                break
            else:
                error = result.get("error", "No data") if isinstance(result, dict) else "Invalid"
                errors.append(f"• {api_name}: {error}")
        except Exception as e:
            errors.append(f"• {api_name}: {str(e)}")

        # Update progress
        try:
            await searching_msg.edit_text(
                f"🔍 *Searching for RC:* `{rc}`\n"
                f"⏳ Attempt {attempted}/{len(WORKING_APIS)}...",
                parse_mode='Markdown'
            )
        except:
            pass

    if data is not None:
        # Format and send vehicle info
        formatted_info = format_vehicle_info(data, rc, used_api)
        await searching_msg.edit_text(formatted_info, parse_mode='Markdown')

        # Save JSON file option
        if raw_data:
            try:
                timestamp = datetime.now().strftime("%Y%m%d_%H%M%S")
                filename = f"ZYNTRASEC_{rc}_{timestamp}.json"
                with open(filename, "w", encoding="utf-8") as f:
                    json.dump(raw_data, f, indent=4, ensure_ascii=False)
                
                # Send JSON file
                with open(filename, "r", encoding="utf-8") as f:
                    json_content = f.read()
                    if len(json_content) < 50000:  # Telegram file size limit
                        await update.message.reply_document(
                            document=(filename, json_content),
                            filename=filename
                        )
                # Clean up
                try:
                    os.remove(filename)
                except:
                    pass
            except Exception as e:
                logger.error(f"Error saving JSON: {e}")

    else:
        # All APIs failed
        error_msg = f"""
❌ *All APIs Failed!*

*RC Number:* `{rc}`

*Errors encountered:*
{chr(10).join(errors) if errors else '• Unknown error'}

*Suggestions:*
• Check internet connection
• RC number might be incorrect
• Try a different RC number
• Contact support: @ZYNTRASEC_Support

🔒 *ZYNTRASEC SECURITY* | © {YEAR}
"""
        await searching_msg.edit_text(error_msg, parse_mode='Markdown')

def main():
    """Main function to run the bot"""
    print("\n" + "="*60)
    print("🔒 ZYNTRASEC SECURITY - Advanced RC Lookup Bot")
    print("="*60)
    print(f"📦 Version: {VERSION}")
    print(f"👤 Author: {AUTHOR}")
    print("="*60 + "\n")
    
    # Setup bot token
    if not setup_bot_token():
        print("❌ Bot token setup failed!")
        sys.exit(1)
    
    # Verify token format
    if not BOT_TOKEN or len(BOT_TOKEN) < 20:
        print("❌ Invalid bot token format!")
        print("   Token should be at least 20 characters long.")
        sys.exit(1)
    
    try:
        # Create application with proper settings
        application = Application.builder().token(BOT_TOKEN).build()
        
        # Add command handlers
        application.add_handler(CommandHandler("start", start_command))
        application.add_handler(CommandHandler("help", help_command))
        application.add_handler(CommandHandler("about", about_command))
        application.add_handler(CommandHandler("status", status_command))
        
        # Add message handler for RC lookup
        application.add_handler(MessageHandler(
            filters.TEXT & ~filters.COMMAND,
            handle_rc_lookup
        ))
        
        # Start the bot
        print("🤖 Starting bot...")
        print(f"🔑 APIs Configured: {len(WORKING_APIS)}")
        print("✅ Bot is running! Press Ctrl+C to stop.\n")
        
        # Run the bot with proper error handling
        application.run_polling(
            drop_pending_updates=True,
            allowed_updates=Update.ALL_TYPES
        )
        
    except KeyboardInterrupt:
        print("\n\n👋 Bot stopped by user")
        sys.exit(0)
    except Exception as e:
        print(f"❌ Error starting bot: {e}")
        print("\n💡 Troubleshooting:")
        print("   1. Check your internet connection")
        print("   2. Verify your bot token is correct")
        print("   3. Make sure the bot is not already running")
        print("   4. Check if Python version is 3.8 or higher")
        print("   5. Try: pip install --upgrade python-telegram-bot")
        sys.exit(1)

if __name__ == "__main__":
    main()
