def install_package(package):
    try:
        subprocess.check_call([sys.executable, '-m', 'pip', 'install', package])
        print(f"✅ Installed: {package}")
        return True
    except Exception as e:
        print(f"❌ Failed to install {package}: {e}")
        return False

def ensure_requirements():
    required = ['playwright', 'requests', 'setuptools']
    for pkg in required:
        try:
            importlib.import_module(pkg)
        except ImportError:
            print(f"⚠️ {pkg} not found. Installing...")
            install_package(pkg)
    try:
        subprocess.check_call([sys.executable, '-m', 'playwright', 'install', 'chromium'])
        print(f"✅ Playwright Chromium Installed.")
    except:
        pass

# 📊 SUBSCRIPTION LOG MANAGEMENT
# ===================================================================
def load_subscription_log():
    if os.path.exists(SUBSCRIPTION_LOG_FILE):
        try:
            with open(SUBSCRIPTION_LOG_FILE, 'r') as f:
                return json.load(f)
        except:
            return {"subscriptions": [], "last_run": None, "subscribed_apis": []}
    return {"subscriptions": [], "last_run": None, "subscribed_apis": []}

def save_subscription_log(log_data):
    with open(SUBSCRIPTION_LOG_FILE, 'w') as f:
        json.dump(log_data, f, indent=4)

def log_subscription(api_name, api_url, status="subscribed"):
    log = load_subscription_log()
    timestamp = datetime.now().isoformat()
    
    # Check if already exists
    for sub in log["subscriptions"]:
        if sub["api_url"] == api_url:
            sub["status"] = status
            sub["last_checked"] = timestamp
            sub["email"] = EMAIL
            save_subscription_log(log)
            return
    
    # New subscription
    log["subscriptions"].append({
        "api_name": api_name,
        "api_url": api_url,
        "status": status,
        "subscribed_at": timestamp,
        "last_checked": timestamp,
        "email": EMAIL
    })
    log["last_run"] = timestamp
    save_subscription_log(log)

def log_api_attempt(api_name, api_url, status="attempted"):
    """Log every API attempt (successful or failed)"""
    log = load_subscription_log()
    timestamp = datetime.now().isoformat()
    
    # Add to attempted APIs list
    if "attempted_apis" not in log:
        log["attempted_apis"] = []
    
    # Check if already attempted
    for api in log["attempted_apis"]:
        if api["api_url"] == api_url:
            api["status"] = status
            api["last_attempt"] = timestamp
            save_subscription_log(log)
            return
    
    log["attempted_apis"].append({
        "api_name": api_name,
        "api_url": api_url,
        "status": status,
        "attempted_at": timestamp,
        "email": EMAIL
    })
    save_subscription_log(log)

# 🧹 DYNAMIC POPUP HANDLER
# ===================================================================
async def handle_popups(page):
    try:
        save_btn = page.locator("button:has-text('Save')").first
        if await save_btn.is_visible():
            await save_btn.click()
            await page.wait_for_timeout(1000)
            print("   🔑 Save password popup: SAVED!")
            return True
        no_thanks_btn = page.locator("button:has-text('No thanks'), button:has-text('Never')").first
        if await no_thanks_btn.is_visible():
            await no_thanks_btn.click()
            await page.wait_for_timeout(1000)
            print("   ❌ Popup removed (No thanks)!")
        cookie_btn = page.locator("button:has-text('Accept All'), button:has-text('Reject All')").first
        if await cookie_btn.is_visible():
            await cookie_btn.click()
            await page.wait_for_timeout(1000)
            print("   🍪 Cookie popup cleared!")
        close_btn = page.locator("button[aria-label='Close'], button:has-text('Close')").first
        if await close_btn.is_visible():
            await close_btn.click()
            await page.wait_for_timeout(1000)
            print("   ❌ Generic popup cleared!")
    except:
        pass
    return False

# ===================================================================
# 🔄 RELOAD PAGE IF ERROR OCCURS
# ===================================================================
async def handle_error_and_reload(page, search_url, max_retries=3):
    for attempt in range(max_retries):
        try:
            error_locator = page.locator("text=Something Went Wrong").first
            if await error_locator.is_visible():
                print(f"   ⚠️ Something Went Wrong page detected! (Attempt {attempt + 1}/{max_retries})")
                
                try:
                    try_again_btn = page.locator("button:has-text('Try again')").first
                    if await try_again_btn.is_visible():
                        await try_again_btn.click()
                        await page.wait_for_timeout(5000)
                        if not await page.locator("text=Something Went Wrong").first.is_visible():
                            print("   ✅ Error resolved after clicking Try again!")
                            return True
                except:
                    pass
                
                print(f"   🔄 Reloading page... (Attempt {attempt + 1}/{max_retries})")
                await page.reload(wait_until="domcontentloaded")
                await page.wait_for_timeout(8000)
                
                if not await page.locator("text=Something Went Wrong").first.is_visible():
                    print(f"   ✅ Error resolved after reload! (Attempt {attempt + 1})")
                    return True
                else:
                    if attempt == max_retries - 1:
                        print("   🔄 Last attempt: Going back to search page...")
                        await page.goto(search_url, wait_until="domcontentloaded")
                        await page.wait_for_timeout(8000)
                        await page.reload(wait_until="domcontentloaded")
                        await page.wait_for_timeout(5000)
                        if not await page.locator("text=Something Went Wrong").first.is_visible():
                            print("   ✅ Error resolved by going to search page!")
                            return True
        except Exception as e:
            print(f"   ⚠️ Error handling attempt {attempt + 1} failed: {e}")
            await page.wait_for_timeout(2000)
    
    print(f"   ❌ Could not resolve error after {max_retries} attempts.")
    return False

async def verify_subscriptions_after_subscribe(page, context, search_url):
    """Verify subscriptions after the main subscription process"""
    
    print(f"\n{Colors.CYAN}  🔍 VERIFYING SUBSCRIPTIONS...{Colors.END}")
    
    try:
        dashboard_url = "https://rapidapi.com/developer/dashboard"
        
        # Go to dashboard
        await page.goto(dashboard_url, wait_until="domcontentloaded")
        await page.wait_for_timeout(5000)
        await handle_error_and_reload(page, dashboard_url, max_retries=3)
        
        print(f"   📊 Looking for subscribed APIs in dashboard...")
        
        subscribed_apis = []
        
        try:
            # Wait for API links
            await page.wait_for_selector("a[href*='/api/']", timeout=15000)
            api_links = page.locator("a[href*='/api/']")
            count = await api_links.count()
            print(f"   ✅ Found {count} API links in dashboard")
            
            for i in range(min(count, 100)):
                try:
                    api_element = api_links.nth(i)
                    api_name = await api_element.text_content() or f"API_{i+1}"
                    api_url = await api_element.get_attribute("href") or ""
                    
                    if api_url:
                        if api_url.startswith("/"):
                            full_url = f"https://rapidapi.com{api_url}"
                        else:
                            full_url = api_url
                        
                        subscribed_apis.append({
                            "name": api_name.strip()[:50],
                            "url": full_url
                        })
                except:
                    continue
                    
        except Exception as e:
            print(f"   ⚠️ Could not find API links: {e}")
            try:
                cards = page.locator(".api-card, .subscribed-api-card")
                count = await cards.count()
                print(f"   ✅ Found {count} API cards")
                
                for i in range(min(count, 100)):
                    try:
                        card = cards.nth(i)
                        text = await card.text_content() or f"API_{i+1}"
                        link = card.locator("a").first
                        if await link.count() > 0:
                            api_url = await link.get_attribute("href") or ""
                            if api_url:
                                if api_url.startswith("/"):
                                    full_url = f"https://rapidapi.com{api_url}"
                                else:
                                    full_url = api_url
                                
                                subscribed_apis.append({
                                    "name": text.strip()[:50],
                                    "url": full_url
                                })
                    except:
                        continue
            except:
                pass

        # Update log
        log = load_subscription_log()
        
        if subscribed_apis:
            # Update status of all existing subscriptions
            current_urls = {sub["api_url"] for sub in log["subscriptions"]}
            
            # Mark all as inactive first
            for sub in log["subscriptions"]:
                sub["status"] = "inactive"
                sub["last_checked"] = datetime.now().isoformat()
            
            # Mark active ones
            for api in subscribed_apis:
                found = False
                for sub in log["subscriptions"]:
                    if sub["api_url"] == api["url"]:
                        sub["status"] = "active"
                        sub["last_checked"] = datetime.now().isoformat()
                        sub["api_name"] = api["name"]
                        found = True
                        break
                
                if not found:
                    # New subscription found
                    log["subscriptions"].append({
                        "api_name": api["name"],
                        "api_url": api["url"],
                        "status": "active",
                        "subscribed_at": datetime.now().isoformat(),
                        "last_checked": datetime.now().isoformat(),
                        "email": EMAIL
                    })
            
            save_subscription_log(log)
        
        # Show verification summary
        log = load_subscription_log()
        active_count = sum(1 for sub in log["subscriptions"] if sub["status"] == "active")
        inactive_count = sum(1 for sub in log["subscriptions"] if sub["status"] == "inactive")
        total_count = len(log["subscriptions"])
        
        print(f"\n{Colors.GREEN}  📊 FINAL VERIFICATION SUMMARY:{Colors.END}")
        print(f"  ─────────────────────────────")
        print(f"  {Colors.GREEN}✅ Active Subscriptions: {active_count}{Colors.END}")
        print(f"  {Colors.YELLOW}⚠️ Inactive: {inactive_count}{Colors.END}")
        print(f"  {Colors.CYAN}📊 Total: {total_count}{Colors.END}")
        
        if active_count > 0:
            print(f"\n  {Colors.GREEN}Active APIs:{Colors.END}")
            for sub in log["subscriptions"]:
                if sub["status"] == "active":
                    print(f"    ✅ {sub['api_name']}")
        
        if inactive_count > 0:
            print(f"\n  {Colors.YELLOW}⚠️ Inactive APIs (Need attention):{Colors.END}")
            for sub in log["subscriptions"]:
                if sub["status"] == "inactive":
                    print(f"    ⚠️ {sub['api_name']}")
            print(f"\n  {Colors.YELLOW}💡 These APIs need to be re-subscribed manually.{Colors.END}")
        
        print(f"\n  {Colors.CYAN}📊 Full report saved in: {SUBSCRIPTION_LOG_FILE}{Colors.END}")
        
    except Exception as e:
        print(f"{Colors.RED}  ❌ Verification error: {e}{Colors.END}")
