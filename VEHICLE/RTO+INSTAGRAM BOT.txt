def cache_result(ttl_func):
    def decorator(func):
        @wraps(func)
        async def wrapper(*args, **kwargs):
            # No global needed – we mutate the dict, not rebind
            ttl = CACHE_TTL
            key = f"{func.__name__}_{args}_{kwargs}"
            key_hash = hashlib.md5(key.encode()).hexdigest()
            if key_hash in cache and (datetime.now() - cache[key_hash]['timestamp']).seconds < ttl:
                stats['cache_hits'] += 1
                return cache[key_hash]['data']
            stats['cache_misses'] += 1
            result = await func(*args, **kwargs)
            cache[key_hash] = {'data': result, 'timestamp': datetime.now()}
            return result
        return wrapper
    return decorator

def retry(max_attempts=3, delay=1, backoff=2, retry_on_status=None):
    if retry_on_status is None:
        retry_on_status = []
    def decorator(func):
        @wraps(func)
        async def wrapper(*args, **kwargs):
            attempt = 0
            current_delay = delay
            while attempt < max_attempts:
                try:
                    return await func(*args, **kwargs)
                except Exception as e:
                    attempt += 1
                    if attempt >= max_attempts:
                        raise
                    logger.warning(f"Retry {attempt}/{max_attempts} for {func.__name__}: {e}")
                    await asyncio.sleep(current_delay)
                    current_delay *= backoff
            return None
        return wrapper
    return decorator

def validate_indian_rc(rc: str) -> bool:
    state_codes = [
        "AP","AR","AS","BR","CG","GA","GJ","HR","HP","JK","JH",
        "KA","KL","MP","MH","MN","ML","MZ","NL","OR","PB","RJ",
        "SK","TN","TR","UP","WB","AN","CH","DN","DD","DL","LD",
        "PY","TS","TG","LA"
    ]
    pattern = r'^(' + '|'.join(state_codes) + r')[-]?\d{1,2}[-]?[A-Z]{1,2}[-]?\d{4}$'
    return re.match(pattern, rc) is not None

@retry(max_attempts=3, retry_on_status=[429, 500, 502, 503, 504])
async def fetch_from_api(identifier: str, api_config: Dict) -> Dict:
    try:
        params = {}
        for k, v in api_config.get("params_mapping", {}).items():
            params[k] = identifier if v is None else v
        loop = asyncio.get_event_loop()
        def _sync_fetch():
            resp = requests.get(api_config["url"], params=params, headers=api_config.get("headers", {}), timeout=10)
            if resp.status_code == 200:
                return resp.json()
            return {"error": f"HTTP {resp.status_code}"}
        return await loop.run_in_executor(None, _sync_fetch)
    except Exception as e:
        return {"error": str(e)}

def flatten_dict(obj, parent='', sep='_'):
    items = {}
    if isinstance(obj, dict):
        for k, v in obj.items():
            new_key = f"{parent}{sep}{k}" if parent else k
            if isinstance(v, (dict, list)):
                items.update(flatten_dict(v, new_key, sep))
            else:
                items[new_key] = v
    elif isinstance(obj, list):
        for i, v in enumerate(obj):
            new_key = f"{parent}{sep}{i}" if parent else str(i)
            if isinstance(v, (dict, list)):
                items.update(flatten_dict(v, new_key, sep))
            else:
                items[new_key] = v
    return items

def normalize_data(raw_data: Dict) -> Dict:
    flat = flatten_dict(raw_data)
    normalized = {}
    for key, val in flat.items():
        clean = key
        for prefix in ['data_', 'result_', 'response_', 'vehicle_', 'rc_', 'user_']:
            if key.startswith(prefix):
                clean = key[len(prefix):]
                break
        normalized[clean] = val
    normalized['_raw'] = raw_data
    return normalized

def split_message(text: str, max_len: int = 4096) -> List[str]:
    if len(text) <= max_len:
        return [text]
    parts = []
    while text:
        if len(text) <= max_len:
            parts.append(text)
            break
        split_idx = text.rfind('\n', 0, max_len)
        if split_idx == -1:
            split_idx = max_len
        parts.append(text[:split_idx])
        text = text[split_idx:].lstrip()
    return parts

def is_private_account(data: Dict) -> bool:
    if 'user' in data and isinstance(data['user'], dict):
        return data['user'].get('is_private', False)
    if 'is_private' in data:
        return data['is_private']
    if 'data' in data and isinstance(data['data'], dict):
        return data['data'].get('is_private', False)
    return False

def extract_all_media(data: Any, depth: int = 0) -> List[Dict]:
    results = []
    if isinstance(data, dict):
        item = extract_single_media(data)
        if item:
            results.append(item)
        for value in data.values():
            results.extend(extract_all_media(value, depth+1))
    elif isinstance(data, list):
        for elem in data:
            results.extend(extract_all_media(elem, depth+1))
    return results

            url = obj['video_url']
            if url:
                media_type = 'video'

    if not url and 'thumbnail' in obj:
        url = obj['thumbnail']
        if url and ('jpg' in url or 'png' in url):
            media_type = 'photo'

    if not url:
        return None

    for cap_key in ['caption', 'caption_text', 'text', 'description']:
        if cap_key in obj and obj[cap_key]:
            caption = str(obj[cap_key])
            break

    likes = obj.get('like_count') or obj.get('likes') or 0
    comments = obj.get('comment_count') or obj.get('comments') or 0
    views = obj.get('view_count') or obj.get('play_count') or obj.get('views') or 0

    ts = obj.get('taken_at') or obj.get('created_at') or obj.get('created_time')
    if ts:
        if isinstance(ts, (int, float)):
            try:
                taken_at = datetime.fromtimestamp(ts).strftime("%d-%b-%Y %H:%M")
            except:
                taken_at = str(ts)
        else:
            taken_at = str(ts)

    meta_parts = []
    if likes: meta_parts.append(f"❤️ {likes}")
    if comments: meta_parts.append(f"💬 {comments}")
    if views: meta_parts.append(f"👁️ {views}")
    if taken_at: meta_parts.append(f"🕒 {taken_at}")
    meta = " | ".join(meta_parts) if meta_parts else ""

    if len(caption) > 200:
        caption = caption[:197] + "..."
    caption_text = f"{caption}\n\n{meta}" if meta else caption

    return {
        'media_type': media_type or 'photo',
        'url': url,
        'caption': caption_text,
        'raw': obj
    }

def extract_media_items_from_response(data: Dict, limit: int = 10) -> List[Dict]:
    all_items = extract_all_media(data)
    if not all_items:
        for key in ['data', 'result', 'items', 'media']:
            if key in data and isinstance(data[key], list):
                for item in data[key]:
                    if isinstance(item, dict):
                        extracted = extract_single_media(item)
                        if extracted:
                            all_items.append(extracted)
        if isinstance(data, list):
            for item in data:
                if isinstance(item, dict):
                    extracted = extract_single_media(item)
                    if extracted:
                        all_items.append(extracted)
    seen = set()
    unique = []
    for item in all_items:
        if item['url'] in seen:
            continue
        seen.add(item['url'])
        unique.append(item)
    return unique[:limit]

@retry(max_attempts=3, retry_on_status=[429])
async def download_media_async(session: aiohttp.ClientSession, url: str, max_size: int = 30 * 1024 * 1024) -> Optional[bytes]:
    try:
        async with session.get(url, timeout=aiohttp.ClientTimeout(total=20)) as resp:
            if resp.status != 200:
                return None
            content_length = resp.headers.get('Content-Length')
            if content_length and int(content_length) > max_size:
                logger.warning(f"File too large: {int(content_length)} bytes")
                return None
            data = bytearray()
            async for chunk in resp.content.iter_chunked(1024*1024):
                data.extend(chunk)
                if len(data) > max_size:
                    return None
            return bytes(data)
    except Exception as e:
        logger.error(f"Download error: {e}")
        return None

async def download_all_media(media_items: List[Dict], progress_callback=None) -> List[Tuple[Dict, Optional[bytes]]]:
    async with aiohttp.ClientSession() as session:
        tasks = []
        for item in media_items:
            tasks.append(download_media_async(session, item['url']))
        results = await asyncio.gather(*tasks, return_exceptions=True)
    final = []
    for i, (item, res) in enumerate(zip(media_items, results)):
        if isinstance(res, Exception) or res is None:
            final.append((item, None))
        else:
            final.append((item, res))
        if progress_callback:
            await progress_callback(i+1, len(media_items))
    return final

        uptime = (datetime.now() - datetime.fromisoformat(stats['start_time'])).total_seconds()
        hours = int(uptime // 3600)
        minutes = int((uptime % 3600) // 60)
        msg = get_text(user_id, 'stats_text',
                       uptime=f"{hours}h {minutes}m",
                       rc=stats['rc_requests'],
                       ig=stats['ig_requests'],
                       hits=stats['cache_hits'],
                       misses=stats['cache_misses'],
                       errors=stats['errors'],
                       cachesize=len(cache))
        await query.edit_message_text(msg, parse_mode="Markdown", reply_markup=get_back_menu())
    elif data == "clearcache":
        cache.clear()
        await query.edit_message_text(get_text(user_id, 'cache_cleared'), reply_markup=get_back_menu())
    elif data == "copy":
        last = user_last.get(user_id)
        if not last:
            await query.edit_message_text(get_text(user_id, 'no_result'), reply_markup=get_back_menu())
        else:
            text = last.get('text', '')
            await query.message.reply_text(f"```\n{text}\n```", parse_mode="Markdown")
            await query.edit_message_text("📋 Result copied above.", reply_markup=get_back_menu())
    elif data == "export":
        last = user_last.get(user_id)
        if not last:
            await query.edit_message_text(get_text(user_id, 'no_result'), reply_markup=get_back_menu())
        else:
            text = last.get('text', '')
            filename = f"result_{user_id}_{datetime.now().strftime('%Y%m%d_%H%M%S')}.txt"
            with open(filename, 'w', encoding='utf-8') as f:
                f.write(text)
            await query.message.reply_document(document=open(filename, 'rb'), caption=get_text(user_id, 'export_file'))
            os.remove(filename)
            await query.edit_message_text("📄 File sent above.", reply_markup=get_back_menu())
    elif data == "refresh":
        await query.edit_message_text("🔄 Refreshed. Use commands.", reply_markup=get_main_menu())
    elif data in ["ig_user_info", "ig_followers", "ig_following", "ig_posts", "ig_reels", "ig_stories", "ig_highlights", "ig_engagement", "ig_suggestions"]:
        cmd_map = {
            "ig_user_info": "user",
            "ig_followers": "followers",
            "ig_following": "following",
            "ig_posts": "posts",
            "ig_reels": "reels",
            "ig_stories": "stories",
            "ig_highlights": "highlights",
            "ig_engagement": "engagement",
            "ig_suggestions": "suggestions",
        }
        cmd = cmd_map.get(data, "")
        await query.edit_message_text(
            f"📷 *{cmd.capitalize()}*\n\nUse `/ig {cmd} <username>`\nExample: `/ig {cmd} _parthu_chaudhari`",
            parse_mode="Markdown",
            reply_markup=get_instagram_menu()
        )
    else:
        await query.edit_message_text("Unknown action.", reply_markup=get_back_menu())

# ---- Error Handler ----
async def error_handler(update: Update, context: ContextTypes.DEFAULT_TYPE):
    logger.error(f"Update {update} caused error {context.error}")
    if update and update.effective_message:
        user_id = update.effective_user.id
        await update.effective_message.reply_text(get_text(user_id, 'error_generic'), reply_markup=get_back_menu())
