class APIQualityAnalyzer:
    def __init__(self):
        self.results = []
        self.best_api = None
        self.best_score = -1
        self.vehicle_number = None
    
    def fetch_from_api(self, rc: str, api_config: dict) -> dict:
        """Fetch data from a single API."""
        try:
            params = {}
            data = None
            json_data = None
            
            if api_config.get("params_mapping"):
                for param_key, param_value in api_config["params_mapping"].items():
                    if param_value is None:
                        params[param_key] = rc
                    else:
                        params[param_key] = param_value
            
            if api_config.get("is_form"):
                data = params
                params = {}
            
            if api_config["method"] == "POST" and not api_config.get("is_form"):
                json_data = params
                params = {}
            
            if api_config["method"] == "GET":
                response = requests.get(
                    api_config["url"],
                    params=params,
                    headers=api_config.get("headers", {}),
                    timeout=15
                )
            elif api_config["method"] == "POST":
                if api_config.get("is_form"):
                    response = requests.post(
                        api_config["url"],
                        data=data,
                        headers=api_config.get("headers", {}),
                        timeout=15
                    )
                else:
                    response = requests.post(
                        api_config["url"],
                        json=json_data,
                        headers=api_config.get("headers", {}),
                        timeout=15
                    )
            else:
                return {"error": f"Unsupported method: {api_config['method']}"}
            
            if response.status_code == 200:
                try:
                    return response.json()
                except:
                    return {"error": "Invalid JSON response"}
            return {"error": f"HTTP {response.status_code}"}
        except Exception as e:
            return {"error": str(e)}
    
    def analyze_data_quality(self, data: dict) -> dict:
        """Analyze the quality of API response data."""
        if not data or "error" in data:
            return {
                "score": 0,
                "fields": 0,
                "has_owner": False,
                "has_address": False,
                "has_vehicle_details": False,
                "has_registration": False,
                "field_names": []
            }
        
        fields = []
        
        # Flatten the data
        flat_data = self.flatten_dict(data)
        
        # Important fields to check
        important_fields = {
            "owner_name": ["owner_name", "owner", "name", "owner_name"],
            "address": ["address", "address_line", "street"],
            "registration_no": ["registration_no", "registration_number", "reg_no", "vehicle_no", "rc_number"],
            "vehicle_class": ["vehicle_class", "class", "vehicle_type"],
            "maker": ["maker", "manufacturer", "make"],
            "model": ["model", "vehicle_model"],
            "fuel_type": ["fuel_type", "fuel", "fuel_type_name"],
            "engine_no": ["engine_no", "engine_number"],
            "chassis_no": ["chassis_no", "chassis_number"],
            "registration_date": ["registration_date", "reg_date", "date"],
            "fitness_upto": ["fitness_upto", "fitness_date"],
            "insurance_upto": ["insurance_upto", "insurance_date"],
            "puc_valid_upto": ["puc_valid_upto", "puc_date"],
            "color": ["color", "vehicle_color"],
            "state": ["state", "state_name", "registered_state"]
        }
        
        # Score calculation
        score = 0
        has_owner = False
        has_address = False
        has_vehicle_details = False
        has_registration = False
        
        # Check each important field
        for category, keywords in important_fields.items():
            found = False
            for key in flat_data:
                for keyword in keywords:
                    if keyword.lower() in key.lower():
                        found = True
                        fields.append(key)
                        break
                if found:
                    break
            
            if found:
                if category == "owner_name":
                    has_owner = True
                    score += 15
                elif category == "address":
                    has_address = True
                    score += 10
                elif category == "registration_no":
                    has_registration = True
                    score += 10
                elif category == "vehicle_class":
                    has_vehicle_details = True
                    score += 8
                elif category in ["maker", "model"]:
                    has_vehicle_details = True
                    score += 5
                elif category in ["fuel_type", "engine_no", "chassis_no"]:
                    has_vehicle_details = True
                    score += 4
                elif category in ["registration_date", "fitness_upto", "insurance_upto", "puc_valid_upto"]:
                    score += 3
                elif category in ["color", "state"]:
                    score += 2
        
        # Additional score for total fields
        total_fields = len(flat_data)
        score += min(total_fields * 0.5, 20)  # Max 20 points for field count
        
        # Bonus for official API (Vahan)
        # This will be added outside
        
        return {
            "score": score,
            "fields": total_fields,
            "has_owner": has_owner,
            "has_address": has_address,
            "has_vehicle_details": has_vehicle_details,
            "has_registration": has_registration,
            "field_names": list(flat_data.keys())[:20]  # Top 20 fields
        }
    
    def flatten_dict(self, obj: dict, parent_key: str = '', sep: str = '_') -> dict:
        """Flatten nested dictionary."""
        items = []
        if isinstance(obj, dict):
            for k, v in obj.items():
                new_key = f"{parent_key}{sep}{k}" if parent_key else k
                if isinstance(v, (dict, list)):
                    items.extend(self.flatten_dict(v, new_key, sep=sep).items())
                else:
                    items.append((new_key, v))
        elif isinstance(obj, list):
            for i, v in enumerate(obj):
                new_key = f"{parent_key}{sep}{i}" if parent_key else str(i)
                if isinstance(v, (dict, list)):
                    items.extend(self.flatten_dict(v, new_key, sep=sep).items())
                else:
                    items.append((new_key, v))
        return dict(items)
    
    def analyze_all_apis(self, rc: str = "MH02FB2727"):
        """Analyze all APIs for quality."""
        print(f"\n{Colors.CYAN}  🔍 Analyzing all 25 APIs for data quality...{Colors.END}")
        print(f"  {Colors.YELLOW}  Vehicle: {rc}{Colors.END}")
        print(f"  {'─'*70}\n")
        
        self.vehicle_number = rc
        results = []
        
        for idx, api_config in enumerate(ALL_APIS, 1):
            api_name = api_config["name"]
            priority = api_config.get("priority", idx)
            
            print(f"  ⏳ [{idx}/{len(ALL_APIS)}] Analyzing: {api_name[:50]}...", end="")
            sys.stdout.flush()
            
            # Fetch data
            data = self.fetch_from_api(rc, api_config)
            
            # Analyze quality
            quality = self.analyze_data_quality(data)
            
            # Add bonus for official API
            if "Vahan" in api_name or "Official" in api_name:
                quality["score"] += 20
            
            # Add priority bonus
            if priority <= 5:
                quality["score"] += 10
            
            result = {
                "priority": priority,
                "name": api_name,
                "status": "Success" if "error" not in data else "Failed",
                "score": quality["score"],
                "fields": quality["fields"],
                "has_owner": quality["has_owner"],
                "has_address": quality["has_address"],
                "has_vehicle_details": quality["has_vehicle_details"],
                "has_registration": quality["has_registration"],
                "field_names": quality["field_names"],
                "data": data if "error" not in data else None
            }
            
            results.append(result)
            
            # Print status
            if "error" in data:
                print(f" {Colors.RED}❌ Failed{Colors.END}")
            else:
                print(f" {Colors.GREEN}✅ Score: {quality['score']}{Colors.END}")
        
        self.results = results
        
        # Find best API
        best = max(results, key=lambda x: x["score"] if x["status"] == "Success" else 0)
        self.best_api = best
        self.best_score = best["score"]
        
        return results

                for key, value in list(flat_data.items())[:10]:
                    count += 1
                    print(f"  {Colors.DIM}  {count}. {key}: {value}{Colors.END}")
        
        # Failed APIs
        failed = [r for r in self.results if r["status"] == "Failed"]
        if failed:
            print(f"\n{Colors.RED}  ❌ Failed APIs ({len(failed)}):{Colors.END}")
            for r in failed:
                print(f"  {Colors.DIM}  - {r['name']}{Colors.END}")
    
    def save_results(self, filename: str = None):
        """Save analysis results to JSON file."""
        if not filename:
            timestamp = datetime.now().strftime("%Y%m%d_%H%M%S")
            filename = f"api_quality_analysis_{self.vehicle_number}_{timestamp}.json"
        
        data = {
            "metadata": {
                "vehicle": self.vehicle_number,
                "total_apis": len(self.results),
                "timestamp": datetime.now().isoformat(),
                "best_api": self.best_api["name"] if self.best_api else None,
                "best_score": self.best_score
            },
            "results": self.results,
            "ranking": sorted(
                [r for r in self.results if r["status"] == "Success"],
                key=lambda x: x["score"],
                reverse=True
            )
        }
        
        with open(filename, "w", encoding="utf-8") as f:
            json.dump(data, f, indent=2, ensure_ascii=False)
        
        print(f"\n  💾 Results saved to: {filename}")
        return filename
