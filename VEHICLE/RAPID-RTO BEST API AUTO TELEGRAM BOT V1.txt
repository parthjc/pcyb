class RateLimitTracker:
    def __init__(self):
        self.rate_limited_apis: Set[str] = set()
        self.rate_limited_urls: Set[str] = set()
        self.rate_limit_cooldown: Dict[str, float] = {}
        self.cooldown_seconds = 60
    
    def is_rate_limited(self, api_name: str, api_url: str) -> bool:
        if api_url in self.rate_limit_cooldown:
            if time.time() < self.rate_limit_cooldown[api_url]:
                return True
            else:
                self.remove_from_rate_limit(api_url)
                return False
        if api_name in self.rate_limited_apis or api_url in self.rate_limited_urls:
            return True
        return False
    
    def mark_rate_limited(self, api_name: str, api_url: str):
        self.rate_limited_apis.add(api_name)
        self.rate_limited_urls.add(api_url)
        self.rate_limit_cooldown[api_url] = time.time() + self.cooldown_seconds
    
    def remove_from_rate_limit(self, api_url: str):
        to_remove = []
        for name in self.rate_limited_apis:
            for api in WORKING_APIS:
                if api.get('name') == name and api.get('url') == api_url:
                    to_remove.append(name)
                    break
        for name in to_remove:
            self.rate_limited_apis.discard(name)
        self.rate_limited_urls.discard(api_url)
        self.rate_limit_cooldown.pop(api_url, None)
    
    def reset_all(self):
        self.rate_limited_apis.clear()
        self.rate_limited_urls.clear()
        self.rate_limit_cooldown.clear()
    
    def get_limited_count(self) -> int:
        return len(self.rate_limited_urls)
    
    def get_limited_list(self) -> List[str]:
        return list(self.rate_limited_apis)

def get_mac_address():
    try:
        mac = subprocess.check_output(['ifconfig', 'eth0'], text=True)
        mac_match = re.search(r'ether\s+([0-9a-fA-F:]{17})', mac)
        if mac_match:
            return mac_match.group(1)
    except:
        pass
    try:
        mac = subprocess.check_output(['ifconfig', 'en0'], text=True)
        mac_match = re.search(r'ether\s+([0-9a-fA-F:]{17})', mac)
        if mac_match:
            return mac_match.group(1)
    except:
        pass
    try:
        mac = subprocess.check_output(['ipconfig', '/all'], text=True)
        mac_match = re.search(r'Physical Address[\s.]+: ([0-9a-fA-F-]{17})', mac)
        if mac_match:
            return mac_match.group(1).replace('-', ':')
    except:
        pass
    try:
        mac = uuid.getnode()
        return ':'.join(['{:02x}'.format((mac >> elements) & 0xff) for elements in range(5, -1, -1)])
    except:
        return None

def auto_set_mac():

def verify_license_file():
    if not os.path.exists(LICENSE_FILE):
        return False
    try:
        with open(LICENSE_FILE, 'r') as f:
            data = json.load(f)
        expected_hash = hashlib.sha256(LICENSE_KEY.encode()).hexdigest()
        if data.get('key_hash') != expected_hash:
            return False
        if data.get('mac'):
            current_mac = get_mac_address()
            if current_mac and current_mac.lower() != data.get('mac', '').lower():
                return False
        if data.get('expiry'):
            current = datetime.now().date()
            expiry = datetime.strptime(data['expiry'], "%Y-%m-%d").date()
            if current > expiry:
                return False
        return True
    except:
        return False

def verify_date():
    try:
        current = datetime.now().date()
        expiry = datetime.strptime(EXPIRY_DATE, "%Y-%m-%d").date()
        return current <= expiry
    except:
        return True

def should_retry(status_code: int, error_text: str = "") -> bool:
    retry_codes = RETRY_CONFIG["retry_status_codes"]
    
    if status_code in retry_codes:
        return True
    
    rate_limit_keywords = ['rate limit', 'too many', 'quota', 'exceeded', '429']
    if any(keyword in error_text.lower() for keyword in rate_limit_keywords):
        return True
    
    service_keywords = ['service unavailable', '503', 'gateway', 'timeout']
    if any(keyword in error_text.lower() for keyword in service_keywords):
        return True
    
    return False

def calculate_backoff(attempt: int) -> float:
    base_delay = RETRY_CONFIG["base_delay"]
    max_delay = RETRY_CONFIG["max_delay"]
    backoff_factor = RETRY_CONFIG["backoff_factor"]
    jitter = RETRY_CONFIG["jitter"]
    
    delay = min(base_delay * (backoff_factor ** attempt), max_delay)
    
    if jitter:
        jitter_amount = delay * 0.3
        delay = delay + random.uniform(-jitter_amount, jitter_amount)
    
    return max(0.5, delay)

def rotate_api_key():
    valid_keys = get_all_valid_keys()
    current_active = get_active_key_id()
    
    if len(valid_keys) <= 1:
        return False
    
    current_index = -1
    for i, (key_id, _) in enumerate(valid_keys):
        if key_id == current_active:
            current_index = i
            break
    
    next_index = (current_index + 1) % len(valid_keys)
    next_key_id = valid_keys[next_index][0]
    
    if set_active_key(next_key_id):
        global RAPIDAPI_KEY
        RAPIDAPI_KEY = get_active_key()
        update_api_headers()
        return True
    
    return False

def fetch_with_retry(rc: str, api_config: Dict, max_retries: int = None, is_auto_mode: bool = False) -> Dict:
    api_name = api_config.get("name", "Unknown")
    api_url = api_config.get("url", "")
    
    if is_auto_mode and is_excluded_from_auto_mode(api_config):
        return {"error": "Excluded from AUTO MODE", "skipped": True}
    
    if max_retries is None:
        max_retries = RETRY_CONFIG["max_retries"]
    
    if is_auto_mode and RETRY_CONFIG["auto_skip_429"]:
        if rate_tracker.is_rate_limited(api_name, api_url):
            return {"error": "Rate limited - auto-skipped", "skipped": True}
    
    attempt = 0
    last_error = None
    original_retries = max_retries
    
    while attempt < max_retries:
        try:
            params = {}
            data = None
            json_data = None
            
            if api_config.get("params_mapping"):
                for param_key, param_value in api_config["params_mapping"].items():
                    if param_value is None:
                        params[param_key] = rc
                    else:
                        params[param_key] = param_value
            
            if api_config.get("is_form"):
                data = params
                params = {}
            
            if api_config["method"] == "POST" and not api_config.get("is_form"):
                json_data = params
                params = {}
            
            timeout = 15 + (attempt * 2)
            
            headers = api_config.get("headers", {}).copy()
            if 'User-Agent' not in headers:
                headers['User-Agent'] = 'Mozilla/5.0 (Windows NT 10.0; Win64; x64) AppleWebKit/537.36 (KHTML, like Gecko) Chrome/120.0.0.0 Safari/537.36'
            
            if api_config["method"] == "GET":
                response = requests.get(
                    api_config["url"],
                    params=params,
                    headers=headers,
                    timeout=timeout
                )
            elif api_config["method"] == "POST":
                if api_config.get("is_form"):
                    response = requests.post(
                        api_config["url"],
                        data=data,
                        headers=headers,
                        timeout=timeout
                    )
                else:
                    response = requests.post(
                        api_config["url"],
                        json=json_data,
                        headers=headers,
                        timeout=timeout
                    )
            else:
                return {"error": f"Unsupported method: {api_config['method']}"}
            
            if response.status_code == 503:
                if attempt >= RETRY_CONFIG["max_503_retries"]:
                    return {"error": f"HTTP 503 - Service Unavailable (max retries)", "status_code": 503}
                
                delay = calculate_backoff(attempt)
                time.sleep(delay)
                attempt += 1
                continue
            
            if response.status_code == 429:
                if is_auto_mode and RETRY_CONFIG["auto_skip_429"]:
                    rate_tracker.mark_rate_limited(api_name, api_url)
                    return {"error": f"HTTP 429 - Rate Limited (Auto-skipped)", "skipped": True}
                
                if RETRY_CONFIG["switch_key_on_429"]:
                    if rotate_api_key():
                        attempt += 1
                        continue
                
                delay = calculate_backoff(attempt)
                time.sleep(delay)
                attempt += 1
                continue
            
            if should_retry(response.status_code, response.text):
                if response.status_code in RETRY_CONFIG["retry_status_codes"]:
                    delay = calculate_backoff(attempt)
                    time.sleep(delay)
                    attempt += 1
                    continue
                
                return {"error": f"HTTP {response.status_code}"}
            
            if response.status_code == 200:
                if rate_tracker.is_rate_limited(api_name, api_url):
                    rate_tracker.remove_from_rate_limit(api_url)
                
                try:
                    return response.json()
                except:
                    return {"raw_text": response.text, "status": "html"}
            
            return {"error": f"HTTP {response.status_code}"}
            
        except requests.exceptions.Timeout:
            delay = calculate_backoff(attempt)
            time.sleep(delay)
            attempt += 1
            last_error = "Timeout"
            
        except requests.exceptions.ConnectionError:
            delay = calculate_backoff(attempt)
            time.sleep(delay)
            attempt += 1
            last_error = "Connection Error"
            
        except Exception as e:
            error_msg = str(e)
            
            if "rate limit" in error_msg.lower() or "too many" in error_msg.lower():
                if is_auto_mode and RETRY_CONFIG["auto_skip_429"]:
                    rate_tracker.mark_rate_limited(api_name, api_url)
                    return {"error": f"Rate Limited - {error_msg}", "skipped": True}
                
                if RETRY_CONFIG["switch_key_on_429"]:
                    if rotate_api_key():
                        attempt += 1
                        continue
            
            delay = calculate_backoff(attempt)
            time.sleep(delay)
            attempt += 1
            last_error = error_msg
    
    return {"error": f"Max retries ({original_retries}) exceeded. Last error: {last_error}"}

def parse_rtosearch_html(html_text: str) -> Dict:
    try:
        soup = BeautifulSoup(html_text, 'html.parser')
        result = {}
        
        patterns = {
            'owner_name': ['Owner Name', 'Owner', 'Name'],
            'registration_no': ['Registration No', 'Vehicle No', 'RC Number'],
            'registration_date': ['Registration Date', 'Reg Date'],
            'vehicle_class': ['Vehicle Class', 'Class'],
            'maker': ['Maker', 'Manufacturer', 'Make'],
            'model': ['Model'],
            'fuel_type': ['Fuel Type', 'Fuel'],
            'engine_no': ['Engine No', 'Engine Number'],
            'chassis_no': ['Chassis No', 'Chassis Number'],
            'insurance_company': ['Insurance Company', 'Insurance'],
            'insurance_upto': ['Insurance Upto', 'Insurance Valid'],
            'color': ['Color'],
            'manufactured_year': ['Manufactured Year', 'Year']
        }
        
        text = soup.get_text()
        lines = [line.strip() for line in text.split('\n') if line.strip()]
        
        for i, line in enumerate(lines):
            for field, keywords in patterns.items():
                for keyword in keywords:
                    if keyword.lower() in line.lower():
                        if ':' in line:
                            parts = line.split(':', 1)
                            if len(parts) == 2:
                                result[field] = parts[1].strip()
                        elif i + 1 < len(lines):
                            result[field] = lines[i + 1].strip()
                        break
        
        tables = soup.find_all('table')
        for table in tables:
            rows = table.find_all('tr')
            for row in rows:
                cells = row.find_all(['td', 'th'])
                if len(cells) >= 2:
                    key = cells[0].get_text().strip()
                    value = cells[1].get_text().strip()
                    for field, keywords in patterns.items():
                        for keyword in keywords:
                            if keyword.lower() in key.lower():
                                result[field] = value
                                break
        
        return result
    except Exception as e:
        return {"error": f"Parse error: {str(e)}"}

def fetch_from_free_api(rc: str, api_config: Dict, is_auto_mode: bool = False) -> Dict:
    result = fetch_with_retry(rc, api_config, is_auto_mode=is_auto_mode)
    
    if "raw_text" in result and result.get("status") == "html":
        html_text = result["raw_text"]
        
        if "rtosearch" in api_config["url"]:
            parsed = parse_rtosearch_html(html_text)
            if parsed and not parsed.get("error"):
                return parsed
        
        return {"raw_text": html_text, "status": "html_parsed"}
    
    return result

def validate_indian_rc(rc: str) -> bool:
    state_codes = [
        "AP","AR","AS","BR","CG","GA","GJ","HR","HP","JK","JH",
        "KA","KL","MP","MH","MN","ML","MZ","NL","OR","PB","RJ",
        "SK","TN","TR","UP","WB","AN","CH","DN","DD","DL","LD",
        "PY","TS","TG","LA"
    ]
    pattern = r'^(' + '|'.join(state_codes) + r')[-]?\d{1,2}[-]?[A-Z]{1,2}[-]?\d{4}$'
    return re.match(pattern, rc) is not None

def flatten_dict(obj: Any, parent_key: str = '', sep: str = '_') -> Dict:
    items = []
    if isinstance(obj, dict):
        for k, v in obj.items():
            new_key = f"{parent_key}{sep}{k}" if parent_key else k
            if isinstance(v, (dict, list)):
                items.extend(flatten_dict(v, new_key, sep=sep).items())
            else:
                items.append((new_key, v))
    elif isinstance(obj, list):
        for i, v in enumerate(obj):
            new_key = f"{parent_key}{sep}{i}" if parent_key else str(i)
            if isinstance(v, (dict, list)):
                items.extend(flatten_dict(v, new_key, sep=sep).items())
            else:
                items.append((new_key, v))
    return dict(items)

def normalize_data(raw_data: Dict) -> Dict:
    flat = flatten_dict(raw_data)
    normalized = {}
    for key, value in flat.items():
        clean_key = key
        for prefix in ['data_', 'result_', 'response_', 'vehicle_', 'rc_']:
            if key.startswith(prefix):
                clean_key = key[len(prefix):]
                break
        normalized[clean_key] = value
    normalized['_raw'] = raw_data
    return normalized

def rc_lookup_with_auto_retry(rc: str, api_config: Dict, is_auto_mode: bool = False) -> Dict:
    valid_keys = get_all_valid_keys()
    
    if not valid_keys:
        return {"error": "No API keys configured!"}
    
    active_key_id = get_active_key_id()
    
    ordered_keys = []
    for key_id, key_value in valid_keys:
        if key_id == active_key_id:
            ordered_keys.insert(0, (key_id, key_value))
        else:
            ordered_keys.append((key_id, key_value))
    
    for key_index, (key_id, key_value) in enumerate(ordered_keys):
        global RAPIDAPI_KEY
        current_key = RAPIDAPI_KEY
        RAPIDAPI_KEY = key_value
        
        if 'headers' in api_config and 'x-rapidapi-key' in api_config['headers']:
            api_config['headers']['x-rapidapi-key'] = key_value
        
        result = fetch_with_retry(rc, api_config, is_auto_mode=is_auto_mode)
        
        RAPIDAPI_KEY = current_key
        if 'headers' in api_config and 'x-rapidapi-key' in api_config['headers']:
            api_config['headers']['x-rapidapi-key'] = current_key
        
        if "error" not in result or "status" in result:
            if "skipped" not in result:
                result['_key_used'] = key_id
            return result
        
        error_msg = result.get("error", "Unknown error")
        
        if result.get("skipped", False):
            return result
        
        if any(keyword in error_msg.lower() for keyword in ['429', 'rate limit', 'too many', 'quota']):
            continue
        
        if any(keyword in error_msg.lower() for keyword in ['503', 'service unavailable']):
            continue
        
        continue
    
    return {"error": "All keys failed after retries"}

def save_results(rc: str, data: Dict, raw: Dict, api_name: str, priority: int, key_used: str = "") -> str:
    timestamp = datetime.now().strftime("%Y%m%d_%H%M%S")
    filename = f"ZYNTRASEC_{rc}_{timestamp}.json"
    data_to_save = {
        "normalized": data,
        "raw": raw,
        "metadata": {
            'tool': 'ZYNTRASEC Security Telegram Bot',
            'version': VERSION,
            'timestamp': datetime.now().isoformat(),
            'rc_checked': rc,
            'api_used': api_name,
            'priority': priority,
            'key_used': key_used,
            'licensed_to': 'ZYNTRASEC SECURITY',
            'expiry_date': EXPIRY_DATE,
            'apis_total': len(WORKING_APIS)
        }
    }
    with open(filename, "w", encoding="utf-8") as f:
        json.dump(data_to_save, f, indent=4, ensure_ascii=False)
    return filename

                reply_markup=reply_markup
            )
            return
        
        elif data.startswith("select_api_"):
            priority_num = int(data.replace("select_api_", ""))
            selected = None
            for api in WORKING_APIS:
                if api.get('priority') == priority_num:
                    selected = api
                    break
            
            if selected:
                context.user_data['selected_api'] = selected
                await query.edit_message_text(
                    f"✅ *Selected API:* {selected['name']}\n\n"
                    f"🔍 Searching for `{context.user_data.get('rc', '')}`...",
                    parse_mode='Markdown'
                )
                # Perform lookup with selected API
                await perform_lookup_with_api(update, context, [selected])
            else:
                await query.edit_message_text("❌ *API not found!*", parse_mode='Markdown')
            return
        
        elif data.startswith("range_"):
            range_parts = data.replace("range_", "").split("_")
            start = int(range_parts[0])
            end = int(range_parts[1])
            
            apis_to_try = []
            for api in WORKING_APIS:
                p = api.get('priority', 0)
                if start <= p <= end:
                    if not rate_tracker.is_rate_limited(api.get('name', ''), api.get('url', '')):
                        apis_to_try.append(api)
            
            if apis_to_try:
                await query.edit_message_text(
                    f"📊 *Selected {len(apis_to_try)} APIs* (Priority {start}-{end})\n\n"
                    f"🔍 Searching for `{context.user_data.get('rc', '')}`...",
                    parse_mode='Markdown'
                )
                await perform_lookup_with_api(update, context, apis_to_try)
            else:
                await query.edit_message_text(
                    "❌ *No APIs available in this range!*",
                    parse_mode='Markdown'
                )
            return
        
        else:
            # Auto mode, top5, free, autokey
            await perform_lookup(update, context)

async def perform_lookup(update: Update, context: ContextTypes.DEFAULT_TYPE) -> None:
    """Perform RC lookup with selected mode."""
    query = update.callback_query
    rc = context.user_data.get('rc', '')
    mode = context.user_data.get('mode', 'auto')
    
    apis_to_try = []
    use_multi_key = False
    is_auto_mode = False
    
    if mode == 'auto':
        apis_to_try = WORKING_APIS
        is_auto_mode = True
    elif mode == 'top5':
        for api in WORKING_APIS:
            if api.get('priority', 0) <= 5:
                if not rate_tracker.is_rate_limited(api.get('name', ''), api.get('url', '')):
                    apis_to_try.append(api)
    elif mode == 'free':
        for api in WORKING_APIS:
            if api.get('is_free', False):
                if not rate_tracker.is_rate_limited(api.get('name', ''), api.get('url', '')):
                    apis_to_try.append(api)
    elif mode == 'autokey':
        apis_to_try = WORKING_APIS
        use_multi_key = True
    
    if not apis_to_try:
        await query.edit_message_text(
            "❌ *No APIs available!*\n\n"
            "All APIs may be rate limited or excluded.\n"
            "Try using 'Free APIs' mode or wait 60 seconds.",
            parse_mode='Markdown'
        )
        return
    
    await perform_lookup_with_api(update, context, apis_to_try, use_multi_key, is_auto_mode)

async def perform_lookup_with_api(update: Update, context: ContextTypes.DEFAULT_TYPE, 
                                   apis_to_try: List, use_multi_key: bool = False, 
                                   is_auto_mode: bool = False) -> None:
    """Perform lookup with specific APIs."""
    query = update.callback_query
    rc = context.user_data.get('rc', '')
    
    data = None
    raw_data = None
    used_api = None
    used_priority = None
    used_key = None
    attempted = 0
    skipped_count = 0
    excluded_count = 0
    service_unavailable_count = 0
    
    status_message = await query.edit_message_text(
        f"🔍 *Searching for {rc}*\n"
        f"📡 Trying {len(apis_to_try)} APIs...\n"
        f"⏳ Progress: 0/{len(apis_to_try)}",
        parse_mode='Markdown'
    )
    
    for i, api_config in enumerate(apis_to_try):
        attempted += 1
        api_name = api_config["name"]
        priority = api_config.get('priority', attempted)
        
        try:
            await status_message.edit_text(
                f"🔍 *Searching for {rc}*\n"
                f"📡 Trying [{i+1}/{len(apis_to_try)}] {api_name[:30]}...\n"
                f"⏳ Priority #{priority}",
                parse_mode='Markdown'
            )
        except:
            pass
        
        if is_auto_mode and is_excluded_from_auto_mode(api_config):
            excluded_count += 1
            continue
        
        if is_auto_mode and RETRY_CONFIG["auto_skip_429"]:
            if rate_tracker.is_rate_limited(api_name, api_config.get('url', '')):
                skipped_count += 1
                continue
        
        if use_multi_key:
            result = rc_lookup_with_auto_retry(rc, api_config, is_auto_mode)
            
            if "error" not in result and len(result) > 0 and "skipped" not in result:
                if "_key_used" in result:
                    key_used = result.pop('_key_used', 'unknown')
                    keys_data, _ = get_all_keys_with_status()
                    key_name = keys_data.get(key_used, {}).get('name', key_used)
                    used_key = f"{key_used} ({key_name})"
                else:
                    used_key = "Default"
                
                raw_data = result
                data = normalize_data(result)
                used_api = api_name
                used_priority = priority
                break
            else:
                error = result.get("error", "No data") if isinstance(result, dict) else "Invalid"
                if result.get("skipped", False):
                    pass
                elif "503" in error or "Service Unavailable" in error:
                    service_unavailable_count += 1
        else:
            if api_config.get('is_free', False):
                result = fetch_from_free_api(rc, api_config, is_auto_mode)
            else:
                result = fetch_with_retry(rc, api_config, is_auto_mode=is_auto_mode)
            
            if "error" not in result and len(result) > 0 and "skipped" not in result:
                raw_data = result
                data = normalize_data(result)
                used_api = api_name
                used_priority = priority
                active_id = get_active_key_id()
                keys_data, _ = get_all_keys_with_status()
                active_name = keys_data.get(active_id, {}).get('name', active_id)
                used_key = active_id + " (" + active_name + ")"
                break
            else:
                error = result.get("error", "No data") if isinstance(result, dict) else "Invalid"
                if result.get("skipped", False):
                    pass
                elif "503" in error or "Service Unavailable" in error:
                    service_unavailable_count += 1
    
    if data is not None:
        result_text = format_vehicle_info(data, rc, used_api, used_priority, used_key)
        await status_message.edit_text(result_text, parse_mode='Markdown')
        
        save_results(rc, data, raw_data if raw_data else {}, used_api, used_priority, used_key)
        
        await context.bot.send_message(
            chat_id=update.effective_chat.id,
            text=f"✅ *Results saved!* 📁 `ZYNTRASEC_{rc}_*.json`",
            parse_mode='Markdown'
        )
    else:
        await status_message.edit_text(
            f"❌ *All APIs failed!*\n\n"
            f"RC: `{rc}`\n"
            f"Attempted: {attempted} APIs\n"
            f"Skipped: {skipped_count} (rate limited)\n"
            f"Excluded: {excluded_count} (Priority 1,2,3)\n"
            f"503 Errors: {service_unavailable_count}\n\n"
            f"💡 Try using 'Free APIs' mode or wait 60 seconds.",
            parse_mode='Markdown'
        )

def format_vehicle_info(data: Dict, rc: str, api_name: str, priority: int, key_used: str = "") -> str:
    state = "Unknown"
    if rc:
        clean_rc = re.sub(r'[-]', '', rc)
        state = clean_rc[:2].upper() if len(clean_rc) >= 2 else "Unknown"
    
    text = f"""
*🚗 VEHICLE INFORMATION*
━━━━━━━━━━━━━━━━━━━━━━━
📋 *RC Number:* `{rc}`
📍 *State:* `{state}`
📡 *API:* {api_name}
🔑 *Key Used:* {key_used}
━━━━━━━━━━━━━━━━━━━━━━━
"""
    
    if "error" in data:
        text += f"\n❌ *Error:* {data['error']}"
        return text
    
    raw_data = data.pop('_raw', None)
    
    field_labels = {
        'owner_name': '👤 Owner Name',
        'owner': '👤 Owner Name',
        'name': '👤 Owner Name',
        'father_name': '👨‍👦 Father Name',
        'address': '📍 Address',
        'city': '🏙️ City',
        'state': '📍 State',
        'state_name': '📍 State',
        'country': '🌍 Country',
        'phone': '📞 Phone Number',
        'mobile': '📞 Phone Number',
        'email': '📧 Email',
        'rto': '🏢 RTO Office',
        'registration_no': '📋 Registration No',
        'vehicle_no': '📋 Registration No',
        'rc_number': '📋 Registration No',
        'registration_date': '📅 Registration Date',
        'vehicle_class': '🚗 Vehicle Class',
        'maker': '🏭 Maker',
        'manufacturer': '🏭 Maker',
        'model': '🔧 Model',
        'fuel_type': '⛽ Fuel Type',
        'engine_no': '🔢 Engine Number',
        'chassis_no': '🔢 Chassis Number',
        'insurance_company': '🛡️ Insurance Company',
        'insurance_upto': '📆 Insurance Upto',
        'puc_valid_upto': '📋 PUC Valid Upto',
        'road_tax_upto': '💰 Road Tax Upto',
        'registration_status': '📊 Registration Status',
        'color': '🎨 Color',
        'manufactured_year': '🏗️ Manufactured Year'
    }
    
    displayed = 0
    for key, value in data.items():
        if key in field_labels:
            label = field_labels.get(key, key.replace('_', ' ').title())
            display_value = value if value else 'N/A'
            text += f"\n{label}: `{display_value}`"
            displayed += 1
        elif not key.startswith('_'):
            for label_key, label_val in field_labels.items():
                if label_key.lower() in key.lower() or key.lower() in label_key.lower():
                    display_value = value if value else 'N/A'
                    text += f"\n{label_val}: `{display_value}`"
                    displayed += 1
                    break
    
    if displayed == 0:
        text += f"\n❌ *No vehicle data found*"
    
    text += f"\n\n━━━━━━━━━━━━━━━━━━━━━━━"
    text += f"\n🔒 ZYNTRASEC SECURITY | {YEAR}"
    text += f"\n📡 Data from: {api_name}"
    
    return text
