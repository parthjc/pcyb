#!/usr/bin/env python3
import os
import sys
import subprocess
import importlib
import asyncio
from playwright.async_api import async_playwright

# ===================================================================
# 📦 AUTO-INSTALL DEPENDENCIES
# ===================================================================
def install_package(package):
    try:
        subprocess.check_call([sys.executable, '-m', 'pip', 'install', package])
        print(f"✅ Installed: {package}")
        return True
    except Exception as e:
        print(f"❌ Failed to install {package}: {e}")
        return False

def ensure_requirements():
    required = ['playwright', 'requests', 'setuptools']
    for pkg in required:
        try:
            importlib.import_module(pkg)
        except ImportError:
            print(f"⚠️ {pkg} not found. Installing...")
            install_package(pkg)
    try:
        subprocess.check_call([sys.executable, '-m', 'playwright', 'install', 'chromium'])
        print(f"✅ Playwright Chromium Installed.")
    except:
        pass

# ===================================================================
# 🎨 COLORS
# ===================================================================
class Colors:
    GREEN = '\033[92m'
    YELLOW = '\033[93m'
    RED = '\033[91m'
    CYAN = '\033[96m'
    MAGENTA = '\033[95m'
    WHITE = '\033[97m'
    BOLD = '\033[1m'
    DIM = '\033[2m'
    END = '\033[0m'

# ================== CONFIGURATION ==================
EMAIL = "ravinar1996@outlook.com"
PASSWORD = "Admin@123"
SEARCH_TERM = "rto"
# ==================================================

async def subscribe_all_apis():
    async with async_playwright() as p:
        # Persistent context: Login cookies save rahese
        user_data_dir = "./user_data"
        
        # ==========================================
        # ⭐ BROWSER SCREEN MEDIUM (1280x720)
        # ==========================================
        context = await p.chromium.launch_persistent_context(
            user_data_dir, 
            headless=False,
            viewport={'width': 1280, 'height': 720}
        )
        page = context.pages[0] if context.pages else await context.new_page()
        page.set_default_timeout(30000)

        print(f"{Colors.CYAN}  🔒 Starting Subscription Process...{Colors.END}")

        # --- STEP 1: LOGIN ---
        print("Opening login page...")
        await page.goto("https://rapidapi.com/auth/login", wait_until="domcontentloaded")
        await page.wait_for_timeout(8000)

        try:
            accept_cookies = page.locator("button:has-text('Accept All')")
            if await accept_cookies.is_visible():
                await accept_cookies.click()
                await page.wait_for_timeout(1000)
                print("✅ Cookie banner accepted.")
        except:
            pass

        if "auth/login" in page.url:
            print("Logging in...")
            try:
                await page.locator("input[type='email']").wait_for(state="visible", timeout=30000)
                await page.locator("input[type='email']").fill(EMAIL)
                await page.locator("input[type='password']").fill(PASSWORD)
                await page.locator("button[type='submit']").click()
                await page.wait_for_timeout(10000)
                print("✅ Login successful!")
            except Exception as e:
                print(f"⚠️ Login issue: {e}")
        else:
            print("✅ Already logged in from previous session!")

        # ==========================================
        # STEP 2: SEARCH + DYNAMIC SCROLL
        # ==========================================
        print(f"\n🔍 Searching for '{SEARCH_TERM}'...")
        search_url = f"https://rapidapi.com/search?term={SEARCH_TERM}"
        await page.goto(search_url, wait_until="domcontentloaded")
        await page.wait_for_timeout(10000)

        # ⬇️ PEHLA MOUSE CURSOR NE NICHE LAVO
        print("   ⬇️ Moving mouse cursor down (to center)...")
        await page.mouse.move(900, 500)
        await page.wait_for_timeout(2000)

        # ⬇️ PACHHI 15 VAAR SCROLL KARO (BADHI APIS LOAD THAY)
        print("   ⬇️ Scrolling down (15 times)...")
        for _ in range(15):
            await page.mouse.wheel(0, 3000)
            await page.wait_for_timeout(1000)

        # ⬇️ PACHHI COUNT KARO
        print("   🔢 Counting ALL APIs...")
        all_api_links = page.locator("a[href*='/api/']")
        total_count = await all_api_links.count()
        print(f"   ✅ Total {total_count} APIs found on the page.\n")

        total_subscribed = 0
        failed_apis = []

        # Badha cards par loop
        for idx in range(total_count):
            print(f"   🖱️ Clicking API {idx+1} of {total_count}...")

            try:
                # Fresh locator
                all_api_links = page.locator("a[href*='/api/']")
                await all_api_links.nth(idx).click()

                # API page load thava mate wait
                await page.wait_for_timeout(8000)

                # Error Handle: "Subscribe to Run" na madiye to "Change plan"
                try:
                    subscribe_run_btn = page.locator("button:has-text('Subscribe to Run')").first
                    await subscribe_run_btn.wait_for(state="visible", timeout=10000)
                    await subscribe_run_btn.click()
                    print(f"      ✅ Clicked 'Subscribe to Run'")
                except:
                    change_plan_btn = page.locator("button:has-text('Change plan')").first
                    await change_plan_btn.wait_for(state="visible", timeout=10000)
                    await change_plan_btn.click()
                    print(f"      ✅ Clicked 'Change plan' (Already Subscribed)")

                # Pricing page load thava mate wait
                await page.wait_for_timeout(5000)

                # Error Handle: "Start Free Plan" na madiye to skip karo
                try:
                    basic_button = page.locator("button:has-text('Start Free Plan'), button:has-text('Request Access')").first
                    await basic_button.wait_for(state="visible", timeout=20000)
                    await basic_button.click()
                    print(f"      ✅ Started Free Basic Plan!")
                except:
                    print(f"      ⚠️ No Free Plan available. Skipping...")
                    failed_apis.append(f"API {idx+1} (No Free Plan)")
                    await page.goto(search_url, wait_until="domcontentloaded")
                    await page.wait_for_timeout(8000)
                    continue

                # Confirm Popup ("OK")
                await page.wait_for_timeout(3000)
                confirm_button = page.locator("button:has-text('OK'), button:has-text('Confirm'), button:has-text('Subscribe')").first
                if await confirm_button.is_visible():
                    await confirm_button.click()
                    print(f"      🎉 SUBSCRIBED! (Total: {total_subscribed + 1})")
                    total_subscribed += 1

                # ==========================================
                # ⬇️ PACHA SEARCH PAGE PAR AAVO (DYNAMIC SCROLL)
                # ==========================================
                await page.goto(search_url, wait_until="domcontentloaded")
                await page.wait_for_timeout(8000)

                # ⬇️ MOUSE CURSOR NE NICHE LAVO (Scroll bar par)
                print(f"   ⬇️ Moving mouse cursor down (to load next API)...")
                await page.mouse.move(900, 500)
                await page.wait_for_timeout(2000)

                # ⬇️ SCROLL KARO (JEM JEM API VADHE TEM TEM SCROLL)
                print(f"   ⬇️ Scrolling down 2-3 times...")
                for _ in range(3):
                    await page.mouse.wheel(0, 3000)
                    await page.wait_for_timeout(1000)

            except Exception as e:
                print(f"      ❌ Error: {e}")
                failed_apis.append(f"API {idx+1} ({e})")
                # Error aave to pan search page par pacha jao
                await page.goto(search_url, wait_until="domcontentloaded")
                await page.wait_for_timeout(8000)
                continue

        # ==========================================
        # STEP 3: FINAL REPORT
        # ==========================================
        print(f"\n📊 FINAL REPORT:")
        print(f"   ✅ Successfully Subscribed: {total_subscribed} APIs")
        print(f"   ❌ Failed APIs: {len(failed_apis)}")
        if failed_apis:
            for api in failed_apis:
                print(f"      - {api}")

        print(f"\n🖥️ Browser 30 second open rahese - Check karo ke badhi subscribed chhe ke nahi.")
        await page.wait_for_timeout(30000)

        await context.close()
        print("Browser closed.")

# ===================================================================
# 📦 INSTALL DEPENDENCIES OPTION
# ===================================================================
def install_requirements():
    print(f"{Colors.CYAN}  📦 Installing/Updating Requirements...{Colors.END}")
    packages = ['playwright', 'requests', 'setuptools']
    for pkg in packages:
        print(f"  ⏳ {pkg}...")
        install_package(pkg)

    print(f"  ⏳ Installing Playwright Chromium Browser...")
    try:
        subprocess.check_call([sys.executable, '-m', 'playwright', 'install', 'chromium'])
    except:
        pass

    print(f"{Colors.GREEN}  ✅ All requirements installed!{Colors.END}")
    input(f"\n{Colors.DIM}Press Enter to continue...{Colors.END}")

# ===================================================================
# 🚀 MAIN MENU
# ===================================================================
def main():
    ensure_requirements()

    clear = lambda: os.system('cls' if os.name == 'nt' else 'clear')
    clear()
    print(f"{Colors.MAGENTA}╔══════════════════════════════════════════════════════════════╗")
    print(f"║   🔒 ZYNTRASEC SECURITY - ALL API SUBSCRIBER                     ║")
    print(f"║   Version: 1.0                                                ║")
    print(f"║   {Colors.END}")
    print(f"╚══════════════════════════════════════════════════════════════╝{Colors.END}")
    print(f"\n{Colors.CYAN}  📋 MENU:{Colors.END}")
    print(f"  {Colors.GREEN}1{Colors.END} 🚀 Subscribe All APIs")
    print(f"  {Colors.GREEN}2{Colors.END} 📦 Install Requirements")
    print(f"  {Colors.GREEN}3{Colors.END} 🚪 Exit")

    choice = input(f"\n{Colors.CYAN}  👉 Enter choice: {Colors.END}").strip()

    if choice == '1':
        asyncio.run(subscribe_all_apis())
    elif choice == '2':
        install_requirements()
        main()
    elif choice == '3':
        print(f"{Colors.GREEN}  👋 Exiting...{Colors.END}")
        sys.exit(0)
    else:
        print(f"{Colors.RED}  ❌ Invalid choice.{Colors.END}")
        main()

if __name__ == "__main__":
    main()
