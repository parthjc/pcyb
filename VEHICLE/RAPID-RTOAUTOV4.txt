def load_api_keys():
    """Load all API keys from file."""
    try:
        if os.path.exists(API_KEYS_FILE):
            with open(API_KEYS_FILE, 'r') as f:
                data = json.load(f)
                if 'keys' not in data:
                    data['keys'] = {}
                for k in ['key1', 'key2', 'key3']:
                    if k not in data['keys']:
                        data['keys'][k] = ""
                if 'active' not in data:
                    data['active'] = 'key1'
                if 'key_names' not in data:
                    data['key_names'] = {
                        "key1": "Primary Key",
                        "key2": "Backup Key 1",
                        "key3": "Backup Key 2"
                    }
                return data
    except:
        pass
    return DEFAULT_KEYS.copy()

def save_api_keys(data):
    """Save all API keys to file."""
    try:
        data['last_updated'] = datetime.now().isoformat()
        with open(API_KEYS_FILE, 'w') as f:
            json.dump(data, f, indent=4)
        return True
    except:
        return False

def get_all_keys_with_status():
    """Get all keys with their status (active/inactive)."""
    data = load_api_keys()
    keys = data.get('keys', {})
    active = data.get('active', 'key1')
    names = data.get('key_names', {})
    
    result = {}
    for key_id in ['key1', 'key2', 'key3']:
        result[key_id] = {
            'key': keys.get(key_id, ''),
            'active': key_id == active,
            'name': names.get(key_id, key_id)
        }
    return result, active

def get_all_valid_keys():
    """Get list of all non-empty keys."""
    data = load_api_keys()
    keys = data.get('keys', {})
    valid_keys = []
    for key_id, key_value in keys.items():
        if key_value and key_value.strip():
            valid_keys.append((key_id, key_value))
    return valid_keys

def set_active_key(key_id):
    """Set which key is active."""
    data = load_api_keys()
    if key_id in data['keys'] and data['keys'][key_id]:
        data['active'] = key_id
        save_api_keys(data)
        return True
    return False

def update_key(key_id, new_key, name=None):
    """Update a specific key."""
    data = load_api_keys()
    if key_id in data['keys']:
        data['keys'][key_id] = new_key.strip()
        if name:
            if 'key_names' not in data:
                data['key_names'] = {}
            data['key_names'][key_id] = name
        save_api_keys(data)
        return True
    return False

def remove_key(key_id):
    """Remove a specific key (set to empty)."""
    data = load_api_keys()
    if key_id in data['keys']:
        data['keys'][key_id] = ""
        if data.get('active') == key_id:
            # Find first non-empty key
            for k, v in data['keys'].items():
                if v:
                    data['active'] = k
                    break
            else:
                data['active'] = 'key1'
        save_api_keys(data)
        return True
    return False

def get_mac_address():
    try:
        mac = subprocess.check_output(['ifconfig', 'eth0'], text=True)
        mac_match = re.search(r'ether\s+([0-9a-fA-F:]{17})', mac)
        if mac_match:
            return mac_match.group(1)
    except:
        pass
    try:
        mac = subprocess.check_output(['ifconfig', 'en0'], text=True)
        mac_match = re.search(r'ether\s+([0-9a-fA-F:]{17})', mac)
        if mac_match:
            return mac_match.group(1)
    except:
        pass
    try:
        mac = subprocess.check_output(['ipconfig', '/all'], text=True)
        mac_match = re.search(r'Physical Address[\s.]+: ([0-9a-fA-F-]{17})', mac)
        if mac_match:
            return mac_match.group(1).replace('-', ':')
    except:
        pass
    try:
        mac = uuid.getnode()
        return ':'.join(['{:02x}'.format((mac >> elements) & 0xff) for elements in range(5, -1, -1)])
    except:
        return None

def auto_set_mac():
    global ALLOWED_MAC
    if not ALLOWED_MAC:
        detected = get_mac_address()
        if detected:
            ALLOWED_MAC = detected
            return True
    return False

def verify_license_file():
    if not os.path.exists(LICENSE_FILE):
        return False
    try:
        with open(LICENSE_FILE, 'r') as f:
            data = json.load(f)
        expected_hash = hashlib.sha256(LICENSE_KEY.encode()).hexdigest()
        if data.get('key_hash') != expected_hash:
            return False
        if data.get('mac'):
            current_mac = get_mac_address()
            if current_mac and current_mac.lower() != data.get('mac', '').lower():
                return False
        if data.get('expiry'):
            current = datetime.now().date()
            expiry = datetime.strptime(data['expiry'], "%Y-%m-%d").date()
            if current > expiry:
                return False
        return True
    except:
        return False

def verify_mac():
    if not ALLOWED_MAC:
        return True
    current_mac = get_mac_address()
    if not current_mac:
        return True
    return current_mac.lower() == ALLOWED_MAC.lower()

        expiry = datetime.strptime(EXPIRY_DATE, "%Y-%m-%d").date()

def add_update_key(key_id):

    if current:
        masked = current[:8] + "..." + current[-4:] if len(current) > 12 else "****"
        print(f"{Colors.GREEN}{masked}{Colors.END} ({current_name})")

        input(f"\n{Colors.DIM}Press Enter to continue...{Colors.END}")
        return

    print(f"\n  {Colors.DIM}💡 Enter a custom name for this key (optional):{Colors.END}")

    print(f"\n  🔄 Testing key...")
    is_valid, message = test_api_key(new_key)

    else:
        print(f"  {Colors.YELLOW}⚠️ {message}{Colors.END}")

def quick_switch_keys():
    """Quickly switch between keys."""
    print(f"\n{Colors.CYAN}  🔄 QUICK SWITCH KEYS (Cycle){Colors.END}")
    print(f"  {'─'*50}")
    
    keys_data, active = get_all_keys_with_status()
    
    # Show available keys
    available = []
    print(f"\n  {Colors.YELLOW}📋 Available Keys:{Colors.END}")
    for key_id, info in keys_data.items():
        key_value = info['key']
        name = info['name']
        if key_value:
            masked = key_value[:8] + "..." + key_value[-4:] if len(key_value) > 12 else "****"
            status = "⭐ ACTIVE" if key_id == active else "  "
            print(f"    {Colors.GREEN}{key_id}{Colors.END}: {masked} {status} {Colors.DIM}({name}){Colors.END}")
            available.append(key_id)
        else:
            print(f"    {Colors.RED}{key_id}{Colors.END}: [EMPTY] {Colors.DIM}({name}){Colors.END}")
    
    if not available:
        print(f"  {Colors.RED}❌ No keys available! Please add keys first.{Colors.END}")
        input(f"\n{Colors.DIM}Press Enter to continue...{Colors.END}")
        return
    
    print(f"\n  {Colors.CYAN}👉 Enter key to switch to (key1, key2, or key3):{Colors.END}")
    print(f"  {Colors.DIM}  (or press Enter to cycle through available keys){Colors.END}")
    choice = input(f"  🔑 Choice: ").strip().lower()
    
    if not choice:
        # Cycle through keys
        current_index = available.index(active) if active in available else 0
        next_index = (current_index + 1) % len(available)
        choice = available[next_index]
        print(f"  🔄 Switching to {choice}...")
    
    if choice in available:
        if set_active_key(choice):
            global RAPIDAPI_KEY
            RAPIDAPI_KEY = keys_data[choice]['key']
            update_api_headers()
            print(f"  {Colors.GREEN}✅ Switched to {choice} ({keys_data[choice]['name']})!{Colors.END}")
        else:
            print(f"  {Colors.RED}❌ Failed to switch!{Colors.END}")
    else:
        print(f"  {Colors.RED}❌ Invalid key or key is empty!{Colors.END}")
    
    input(f"\n{Colors.DIM}Press Enter to continue...{Colors.END}")

def test_all_keys():
    """Test all saved keys."""
    print(f"\n{Colors.CYAN}  🧪 TEST ALL KEYS{Colors.END}")
    print(f"  {'─'*50}")
    
    keys_data, active = get_all_keys_with_status()
    
    print(f"\n  🔄 Testing all keys...\n")
    
    results = []
    for key_id, info in keys_data.items():
        key_value = info['key']
        name = info['name']
        if key_value:
            masked = key_value[:8] + "..." + key_value[-4:] if len(key_value) > 12 else "****"
            print(f"  Testing {Colors.GREEN}{key_id}{Colors.END}: {masked} {Colors.DIM}({name}){Colors.END}")
            is_valid, message = test_api_key(key_value)
            if is_valid:
                print(f"    {Colors.GREEN}✅ {message}{Colors.END}")
                results.append((key_id, True, message))
            else:
                print(f"    {Colors.RED}❌ {message}{Colors.END}")
                results.append((key_id, False, message))
        else:
            print(f"  Testing {Colors.RED}{key_id}{Colors.END}: [EMPTY] {Colors.DIM}({name}) - Skipped{Colors.END}")
    
    # Summary
    print(f"\n  {Colors.CYAN}📊 SUMMARY:{Colors.END}")
    print(f"  {'─'*50}")
    valid_count = sum(1 for r in results if r[1])
    total_count = len(results)
    print(f"  ✅ Valid: {valid_count}/{total_count}")
    print(f"  ❌ Invalid: {total_count - valid_count}/{total_count}")
    
    if results:
        print(f"\n  {Colors.YELLOW}📋 Details:{Colors.END}")
        for key_id, is_valid, message in results:
            status = Colors.GREEN + "✅" + Colors.END if is_valid else Colors.RED + "❌" + Colors.END
            print(f"    {key_id}: {status} {message}")
    
    print(f"\n  {Colors.GREEN}✅ Testing complete!{Colors.END}")
    input(f"\n{Colors.DIM}Press Enter to continue...{Colors.END}")

def validate_indian_rc(rc: str) -> bool:
    state_codes = [
        "AP","AR","AS","BR","CG","GA","GJ","HR","HP","JK","JH",
        "KA","KL","MP","MH","MN","ML","MZ","NL","OR","PB","RJ",
        "SK","TN","TR","UP","WB","AN","CH","DN","DD","DL","LD",
        "PY","TS","TG","LA"
    ]
    pattern = r'^(' + '|'.join(state_codes) + r')[-]?\d{1,2}[-]?[A-Z]{1,2}[-]?\d{4}$'
    return re.match(pattern, rc) is not None

def fetch_from_api(rc: str, api_config: Dict) -> Dict:
    try:
        params = {}
        data = None
        json_data = None
        
        if api_config.get("params_mapping"):
            for param_key, param_value in api_config["params_mapping"].items():
                if param_value is None:
                    params[param_key] = rc
                else:
                    params[param_key] = param_value
        
        if api_config.get("is_form"):
            data = params
            params = {}
        
        if api_config["method"] == "POST" and not api_config.get("is_form"):
            json_data = params
            params = {}
        
        if api_config["method"] == "GET":
            response = requests.get(
                api_config["url"],
                params=params,
                headers=api_config.get("headers", {}),
                timeout=15
            )
        elif api_config["method"] == "POST":
            if api_config.get("is_form"):
                response = requests.post(
                    api_config["url"],
                    data=data,
                    headers=api_config.get("headers", {}),
                    timeout=15
                )
            else:
                response = requests.post(
                    api_config["url"],
                    json=json_data,
                    headers=api_config.get("headers", {}),
                    timeout=15
                )
        else:
            return {"error": f"Unsupported method: {api_config['method']}"}
        
        if response.status_code == 200:
            try:
                return response.json()
            except:
                return {"raw_text": response.text, "status": "non-json"}
        return {"error": f"HTTP {response.status_code}"}
    except Exception as e:
        return {"error": str(e)}

def flatten_dict(obj: Any, parent_key: str = '', sep: str = '_') -> Dict:
    items = []
    if isinstance(obj, dict):
        for k, v in obj.items():
            new_key = f"{parent_key}{sep}{k}" if parent_key else k
            if isinstance(v, (dict, list)):
                items.extend(flatten_dict(v, new_key, sep=sep).items())
            else:
                items.append((new_key, v))
    elif isinstance(obj, list):
        for i, v in enumerate(obj):
            new_key = f"{parent_key}{sep}{i}" if parent_key else str(i)
            if isinstance(v, (dict, list)):
                items.extend(flatten_dict(v, new_key, sep=sep).items())
            else:
                items.append((new_key, v))
    return dict(items)

def normalize_data(raw_data: Dict) -> Dict:
    flat = flatten_dict(raw_data)
    normalized = {}
    for key, value in flat.items():
        clean_key = key
        for prefix in ['data_', 'result_', 'response_', 'vehicle_', 'rc_']:
            if key.startswith(prefix):
                clean_key = key[len(prefix):]
                break
        normalized[clean_key] = value
    normalized['_raw'] = raw_data
    return normalized

            for key_id, key_value in valid_keys:
                keys_data, _ = get_all_keys_with_status()
                name = keys_data.get(key_id, {}).get('name', key_id)
                masked = key_value[:8] + "..." + key_value[-4:] if len(key_value) > 12 else "****"
                print(f"     {Colors.CYAN}{key_id}{Colors.END}: {masked} {Colors.DIM}({name}){Colors.END}")
        
        else:
            print(f"{Colors.RED}  ❌ Invalid mode!{Colors.END}")
            input(f"\n{Colors.DIM}Press Enter to retry...{Colors.END}")
            continue
        
        if not apis_to_try:
            print(f"{Colors.RED}  ❌ No APIs selected!{Colors.END}")
            input(f"\n{Colors.DIM}Press Enter to retry...{Colors.END}")
            continue

        # ============================================================
        # 🔍 SEARCH WITH SELECTED APIS
        # ============================================================
        print(f"\n  🔄 Searching for {rc} with {len(apis_to_try)} API(s)...")
        if use_multi_key_fallback:
            print(f"  🔑 Will auto-retry with all available keys if needed\n")
        loading_animation()

        data = None
        raw_data = None
        used_api = None
        used_priority = None
        used_key = None
        attempted = 0
        start_time = time.time()

        for api_config in apis_to_try:
            attempted += 1
            api_name = api_config["name"]
            priority = api_config.get('priority', attempted)
            
            if use_multi_key_fallback:
                # Try with all keys
                print(f"  ⏳ [{attempted}/{len(apis_to_try)}] Trying {api_name} with ALL keys...")
                
                result = try_with_all_keys(rc, api_config)
                
                if "error" not in result and len(result) > 0 and "_key_used" in result:
                    key_used = result.pop('_key_used', 'unknown')
                    keys_data, _ = get_all_keys_with_status()
                    key_name = keys_data.get(key_used, {}).get('name', key_used)
                    raw_data = result
                    data = normalize_data(result)
                    used_api = api_name
                    used_priority = priority
                    used_key = f"{key_used} ({key_name})"
                    print(f"  {Colors.GREEN}✅ Success with {used_key}! (Priority #{priority}){Colors.END}")
                    break
                else:
                    error = result.get("error", "No data") if isinstance(result, dict) else "Invalid"
                    print(f"  {Colors.RED}❌ All keys failed: {error}{Colors.END}")
                    time.sleep(0.1)
            else:
                # Try with single key (active)
                print(f"  ⏳ [{attempted}/{len(apis_to_try)}] Trying {api_name}...")
                
                result = fetch_from_api(rc, api_config)

                if "error" not in result and len(result) > 0:
                    raw_data = result
                    data = normalize_data(result)
                    used_api = api_name
                    used_priority = priority
                    active_id = get_active_key_id()
                    keys_data, _ = get_all_keys_with_status()
                    active_name = keys_data.get(active_id, {}).get('name', active_id)
                    used_key = active_id + " (" + active_name + ")"
                    print(f"  {Colors.GREEN}✅ Success! (Priority #{priority}){Colors.END}")
                    break
                else:
                    error = result.get("error", "No data") if isinstance(result, dict) else "Invalid"
                    print(f"  {Colors.RED}❌ {error}{Colors.END}")
                    time.sleep(0.1)

        elapsed_time = time.time() - start_time

        if data is not None:
            print(f"\n  ⏱️ Time taken: {elapsed_time:.2f} seconds")
            display_vehicle_info(data, rc, used_api, used_priority, used_key)
            save_results(rc, data, raw_data if raw_data else {}, used_api, used_priority, used_key)
            print(f"\n  {Colors.GREEN}✅ All data fetched successfully!{Colors.END}")
        else:
            print(f"\n  {Colors.RED}❌ All {len(apis_to_try)} APIs failed!{Colors.END}")
            print(f"  {Colors.YELLOW}💡 Check internet or try another RC{Colors.END}")

        print("\n  Press Enter to continue or type 'menu' to go back")
        choice = input().strip().lower()
        if choice == 'menu':
            return
