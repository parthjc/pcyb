def get_mac_address():
    try:
        mac = subprocess.check_output(['ifconfig', 'eth0'], text=True)
        mac_match = re.search(r'ether\s+([0-9a-fA-F:]{17})', mac)
        if mac_match:
            return mac_match.group(1)
    except:
        pass
    try:
        mac = subprocess.check_output(['ifconfig', 'en0'], text=True)
        mac_match = re.search(r'ether\s+([0-9a-fA-F:]{17})', mac)
        if mac_match:
            return mac_match.group(1)
    except:
        pass
    try:
        mac = subprocess.check_output(['ipconfig', '/all'], text=True)
        mac_match = re.search(r'Physical Address[\s.]+: ([0-9a-fA-F-]{17})', mac)
        if mac_match:
            return mac_match.group(1).replace('-', ':')
    except:
        pass
    try:
        mac = uuid.getnode()
        return ':'.join(['{:02x}'.format((mac >> elements) & 0xff) for elements in range(5, -1, -1)])
    except:
        return None

def auto_set_mac():
    global ALLOWED_MAC
    if not ALLOWED_MAC:
        detected = get_mac_address()
        if detected:
            ALLOWED_MAC = detected
            return True
    return False

def verify_license_key():
    print(f"{Colors.CYAN}  🔒 Enter license key to continue{Colors.END}")
    entered = input(f"{Colors.MAGENTA}  🔑 License Key: {Colors.END}").strip()
    expected_hash = hashlib.sha256(LICENSE_KEY.encode()).hexdigest()
    entered_hash = hashlib.sha256(entered.encode()).hexdigest()
    if entered_hash == expected_hash:
        print(f"{Colors.GREEN}  ✅ License verified!{Colors.END}")
        return True
    else:
        print(f"{Colors.RED}  ❌ Invalid license key!{Colors.END}")
        return False

def verify_license_file():
    if not os.path.exists(LICENSE_FILE):
        return False
    try:
        with open(LICENSE_FILE, 'r') as f:
            data = json.load(f)
        expected_hash = hashlib.sha256(LICENSE_KEY.encode()).hexdigest()
        if data.get('key_hash') != expected_hash:
            return False
        if data.get('mac'):
            current_mac = get_mac_address()
            if current_mac and current_mac.lower() != data.get('mac', '').lower():
                return False
        if data.get('expiry'):
            current = datetime.now().date()
            expiry = datetime.strptime(data['expiry'], "%Y-%m-%d").date()
            if current > expiry:
                return False
        return True
    except:
        return False

def verify_mac():
    if not ALLOWED_MAC:
        return True
    current_mac = get_mac_address()
    if not current_mac:
        return True
    return current_mac.lower() == ALLOWED_MAC.lower()

def verify_date():
    try:
        current = datetime.now().date()
        expiry = datetime.strptime(EXPIRY_DATE, "%Y-%m-%d").date()
        return current <= expiry
    except:
        return True

def validate_indian_rc(rc: str) -> bool:
    state_codes = [
        "AP","AR","AS","BR","CG","GA","GJ","HR","HP","JK","JH",
        "KA","KL","MP","MH","MN","ML","MZ","NL","OR","PB","RJ",
        "SK","TN","TR","UP","WB","AN","CH","DN","DD","DL","LD",
        "PY","TS","TG","LA"
    ]
    pattern = r'^(' + '|'.join(state_codes) + r')[-]?\d{1,2}[-]?[A-Z]{1,2}[-]?\d{4}$'
    return re.match(pattern, rc) is not None

def fetch_from_api(rc: str, api_config: Dict) -> Dict:
    try:
        params = {}
        if api_config.get("params_mapping"):
            for param_key, _ in api_config["params_mapping"].items():
                params[param_key] = rc
        response = requests.get(
            api_config["url"],
            params=params,
            headers=api_config.get("headers", {}),
            timeout=10
        )
        if response.status_code == 200:
            return response.json()
        return {"error": f"HTTP {response.status_code}"}
    except Exception as e:
        return {"error": str(e)}

def flatten_dict(obj: Any, parent_key: str = '', sep: str = '_') -> Dict:
    items = []
    if isinstance(obj, dict):
        for k, v in obj.items():
            new_key = f"{parent_key}{sep}{k}" if parent_key else k
            if isinstance(v, (dict, list)):
                items.extend(flatten_dict(v, new_key, sep=sep).items())
            else:
                items.append((new_key, v))
    elif isinstance(obj, list):
        for i, v in enumerate(obj):
            new_key = f"{parent_key}{sep}{i}" if parent_key else str(i)
            if isinstance(v, (dict, list)):
                items.extend(flatten_dict(v, new_key, sep=sep).items())
            else:
                items.append((new_key, v))
    return dict(items)

def normalize_data(raw_data: Dict) -> Dict:
    flat = flatten_dict(raw_data)
    normalized = {}
    for key, value in flat.items():
        clean_key = key
        for prefix in ['data_', 'result_', 'response_', 'vehicle_', 'rc_']:
            if key.startswith(prefix):
                clean_key = key[len(prefix):]
                break
        normalized[clean_key] = value
    normalized['_raw'] = raw_data
    return normalized

def rc_lookup():
    """Main RC lookup function."""
    clear_screen()
    print_banner()
    print(f"{Colors.CYAN}  🔍 Enter Indian RC Number{Colors.END}")
    print(f"{Colors.YELLOW}  📝 Format: GJ-05-VC-1234 or GJ05VC1234{Colors.END}")
    print(f"{Colors.DIM}  💡 Press Enter to go back to menu{Colors.END}\n")
    
    while True:
        rc = input(f"{Colors.GREEN}  🚗 RC Number (or Enter to go back): {Colors.END}").strip().upper()
        if not rc:
            return
        
        if not validate_indian_rc(rc):
            print(f"{Colors.YELLOW}  ⚠️ Not a valid Indian RC format{Colors.END}")
            confirm = input("  Continue anyway? (y/n): ").strip().lower()
            if confirm != 'y':
                continue

        print(f"\n  🔄 Searching for {rc}...")
        loading_animation()

        data = None
        raw_data = None
        used_api = None
        attempted = 0

        for api_config in WORKING_APIS:
            attempted += 1
            api_name = api_config["name"]
            print(f"  ⏳ [{attempted}/{len(WORKING_APIS)}] Trying {api_name}...")
            result = fetch_from_api(rc, api_config)

            if "error" not in result and len(result) > 0:
                raw_data = result
                data = normalize_data(result)
                used_api = api_name
                print(f"  {Colors.GREEN}✅ Success!{Colors.END}")
                break
            else:
                error = result.get("error", "No data") if isinstance(result, dict) else "Invalid"
                print(f"  {Colors.RED}❌ {error}{Colors.END}")
                time.sleep(0.2)

        if data is not None:
            display_vehicle_info(data, rc, used_api)
            save_results(rc, data, raw_data if raw_data else {})
            print(f"\n  {Colors.GREEN}✅ All data fetched successfully!{Colors.END}")
        else:
            print(f"\n  {Colors.RED}❌ All APIs failed!{Colors.END}")
            print(f"  {Colors.YELLOW}💡 Check internet or try another RC{Colors.END}")

        print("\n  Press Enter to continue or type 'menu' to go back")
        choice = input().strip().lower()
        if choice == 'menu':
            return
